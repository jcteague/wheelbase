# Combined watchlist — A · Overview

Compare qualifying contracts across stocks, with the rest of your watchlist directly below.

Use the preview controls for no matches and an outage; add a stock, try a duplicate ticker, select stock details, or preview the trade handoff. No data is saved outside this mockup.

export const addSchema = z.object({ ticker: z.string().trim().toUpperCase().regex(/^[A-Z]{1,5}$/, 'Enter a ticker of 1–5 letters'), note: z.string().trim().max(500) })

export function AddStock({ entries, onAdd, onCancel }) {
  const { register, handleSubmit, setError, formState: { errors } } = useForm({ resolver: zodResolver(addSchema), defaultValues: { ticker: '', note: '' } })
  return (
    <SectionCard header="Add a stock to watch">
      <form className="flex flex-col gap-4 p-5" onSubmit={handleSubmit(values => {
        if (entries.some(s => s.ticker === values.ticker)) { setError('ticker', { message: values.ticker + ' is already on your watchlist' }); return }
        onAdd({ ticker: values.ticker, name: 'New stock', price: '—', change: '—', ivr: '—', ready: false, reason: 'Not screened yet', note: values.note || 'No thesis yet.', conditions: ['No personal conditions'], earnings: 'Not checked' })
      })}>
        <div className="grid gap-4 md:grid-cols-[160px_1fr]">
          <Field label="Ticker" htmlFor="add-ticker" error={errors.ticker?.message}><Input id="add-ticker" placeholder="e.g. AMZN" {...register('ticker')} /></Field>
          <Field label="Why are you interested?" htmlFor="add-note" error={errors.note?.message}><Input id="add-note" placeholder="Optional thesis or research note" {...register('note')} /></Field>
        </div>
        <p className="text-xs text-wb-text-muted">New stocks start in Stocks of interest until screened. Entry-condition editing is a separate interaction in this concept.</p>
        <div className="flex gap-2"><Button type="submit" size="sm">Add to watchlist</Button><Button type="button" variant="ghost" size="sm" onClick={onCancel}>Cancel</Button></div>
      </form>
    </SectionCard>
  )
}

export function StockDetail({ stock, onReview }) {
  return (
    <div className="flex flex-col gap-5 p-5">
      <div className="flex items-start justify-between gap-3">
        <div><p className="font-wb-mono text-xs tracking-widest text-wb-text-muted">STOCK DETAIL</p><h3 className="mt-2 font-wb-mono text-3xl font-semibold text-wb-gold">{stock.ticker}</h3><p className="mt-1 text-sm text-wb-text-secondary">{stock.name}</p></div>
        <Badge color={stock.ready ? 'var(--wb-green)' : undefined}>{stock.ready ? 'Meets criteria' : 'Watching'}</Badge>
      </div>
      <div className="grid grid-cols-3 gap-3 rounded-md bg-wb-bg-elevated p-4">
        {[['Last price', stock.price], ['Day change', stock.change], ['IV rank', stock.ivr]].map(([label, value]) => <div key={label}><p className="text-xs text-wb-text-muted">{label}</p><p className="mt-1 font-wb-mono text-lg">{value}</p></div>)}
      </div>
      <div><h4 className="mb-2 text-xs font-semibold uppercase tracking-widest text-wb-text-muted">Your thesis</h4><p className="text-sm leading-relaxed text-wb-text-secondary">{stock.note}</p></div>
      <div><h4 className="mb-2 text-xs font-semibold uppercase tracking-widest text-wb-text-muted">Entry conditions</h4><div className="flex flex-wrap gap-2">{stock.conditions.map(c => <Badge key={c}>{c}</Badge>)}</div></div>
      <div className="flex justify-between gap-3 text-xs"><span className="text-wb-text-muted">Earnings</span><span>{stock.earnings}</span></div>
      {stock.ready ? (
        <SectionCard header="Matching put · best score for this stock">
          <div className="flex flex-col gap-4 p-4">
            <div className="flex items-center justify-between"><span className="font-wb-mono text-xl">{stock.strike} PUT</span><span className="text-sm text-wb-text-secondary">{stock.expiry} · {stock.dte} DTE</span></div>
            <dl className="grid grid-cols-3 gap-4">{[['Mark / share', stock.mark], ['Period yield', stock.period], ['Annualized', stock.annual], ['Delta', stock.delta], ['Open interest', stock.oi], ['Spread', stock.spread]].map(([label, value]) => <div key={label}><dt className="text-xs text-wb-text-muted">{label}</dt><dd className="mt-1 font-wb-mono text-sm">{value}</dd></div>)}</dl>
            <p className="text-xs text-wb-text-muted">Cash to secure 1 contract: {stock.cash}. Yield uses mark ÷ strike, before fees.</p>
            <Button size="sm" onClick={() => onReview(stock)}>Review trade →</Button>
          </div>
        </SectionCard>
      ) : <AlertBox variant="warning">{stock.reason}. This stock stays on your watchlist while you wait.</AlertBox>}
    </div>
  )
}
