import { useState } from 'react'

# US-61: Expiring Soon Flags

_As a wheel trader scanning the dashboard and calendar, I want positions expiring within 7 days to stand out before they become critical._

## Acceptance Criteria States

- **dashboard**: row-level gold DTE flag on the dashboard surface
- **calendar**: matching visual emphasis on the expiration calendar chip and date detail
- **clean**: positions at 8 DTE or more render without the extra emphasis

---

export const MONO = "'JetBrains Mono', 'Fira Mono', monospace"

export function GoldFlag({ label }) {
  return (
    <span
      style={{
        padding: '4px 8px',
        borderRadius: 999,
        background: 'rgba(230,168,23,0.14)',
        border: '1px solid rgba(230,168,23,0.3)',
        color: '#f6c445',
        fontSize: '0.66rem',
        fontWeight: 700,
        letterSpacing: '0.08em'
      }}
    >
      {label}
    </span>
  )
}
