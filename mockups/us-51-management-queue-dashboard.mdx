# US-51: Management Queue Dashboard

_As a wheel trader starting my day, I want a management queue at the top of the dashboard ordered by urgency, so that I know what needs attention first._

## Acceptance Criteria States

- **queue**: top-of-dashboard queue with three alerts sorted high to low, each showing ticker, phase, trigger summary, and a quick action
- **dense**: same surface with multiple open alerts and a visibly crowded week to confirm the list remains scannable
- **empty**: calm empty state when no positions currently need attention

---

export function UrgencyPill({ level }) {
  const map = {
    HIGH: { fg: '#f85149', bg: 'rgba(248,81,73,0.12)' },
    MED: { fg: '#e6a817', bg: 'rgba(230,168,23,0.12)' },
    LOW: { fg: '#58a6ff', bg: 'rgba(88,166,255,0.12)' }
  }
  const tone = map[level]
  return (
    <span
      style={{
        padding: '3px 8px',
        borderRadius: 999,
        background: tone.bg,
        color: tone.fg,
        fontSize: '0.65rem',
        fontWeight: 700,
        letterSpacing: '0.08em'
      }}
    >
      {level}
    </span>
  )
}

export function PhasePill({ phase }) {
  const map = {
    CSP_OPEN: '#e6a817',
    CC_OPEN: '#bc8cff'
  }
  return (
    <span
      style={{
        color: map[phase] ?? '#9fb0c0',
        fontSize: '0.72rem',
        fontWeight: 600,
        letterSpacing: '0.08em'
      }}
    >
      {phase.replace('_', ' ')}
    </span>
  )
}

export function QueueCard({ item }) {

}
