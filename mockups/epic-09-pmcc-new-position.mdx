# Epic 09: New Position Sheet — Final Layout

_The **New position** sheet as it looks once every Epic 09 story that touches it has shipped, rendered over the shipped positions list. One 460px sheet, two strategies: **Standard** is the wheel form exactly as it ships today, **PMCC** stacks a Buy LEAPS call section, a Sell short call section, the IV term-structure read and the opening cash flows. Every figure on the sheet — DTE, cash flows, debit-to-width, IV spread and ratio, the promote banner's "net debit at fresh marks" — is folded from the fixture in this file, never typed in._

## Which story owns what

The chrome is imported, not redrawn: `PageHeader`, `MarketStatusPill`, `Badge`, `SheetOverlay` / `SheetPanel` / `SheetHeader` / `SheetBody` / `SheetFooter`, `Field`, `NumberInput`, `DatePicker`, `FormButton`, `SectionCard`, `AlertBox` and `ErrorAlert` all come from `src/renderer/src`.

export const ms = (s) => Date.parse(`${s}T00:00:00Z`)

export const days = (a, b) => Math.round((ms(b) - ms(a)) / DAY)

export const money = (n) =>
  `${n < 0 ? '−' : ''}$${Math.abs(n).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`

export const fix2 = (n) => n.toFixed(2)

export const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec']

export const longDate = (iso) => {
  const [y, m, d] = iso.split('-').map(Number)
  return `${MONTHS[m - 1]} ${d}, ${y}`
}

export const pct2 = (x) => `${(x * 100).toFixed(2)}%`

export const ivPct = (iv) => (iv == null ? '—' : `${(iv * 100).toFixed(1)}%`)

export const num = (s) => {
  if (s === '' || s == null) return null
  const n = Number(s)
  return Number.isFinite(n) ? n : null
}

export function readTermStructure({ shortIv, leapsIv, shortSource, leapsSource, quoteState }) {
  // Withheld states short-circuit the registry in US-113's order:
  // unavailable → stale → manual → missing IV. Figures stay muted where they exist.
  if (quoteState === 'unavailable') {
    return { kind: 'withheld', status: 'Quotes unavailable · —', reason: 'no live quote to read IV from' }
  }
  if (quoteState === 'stale') {
    return {
      kind: 'withheld',
      status: 'Stale · —',
      reason: 'a quote is stale — verify against a live quote before reading it'
    }
  }
  if (shortSource === 'manual') {
    return { kind: 'withheld', status: 'n/a', reason: 'manual entry — no quote to read the short call’s IV' }
  }
  if (leapsSource === 'manual') {
    return { kind: 'withheld', status: 'n/a', reason: 'manual entry — no quote to read the LEAPS IV' }
  }
  const shortMissing = shortIv == null
  const leapsMissing = leapsIv == null || leapsIv === 0
  if (shortMissing || leapsMissing) {
    const reason =
      shortMissing && leapsMissing
        ? 'IV not quoted for either call'
        : shortMissing
          ? 'IV not quoted for the short call'
          : 'IV not quoted for the LEAPS'
    return { kind: 'withheld', status: 'n/a', reason }
  }
  const ratio = shortIv / leapsIv
  const spreadPts = (shortIv - leapsIv) * 100
  const read = IV_READS.find((r) => r.test(ratio))
  return { kind: 'read', ...read, ratio, spreadPts }
}

export const fmtSpread = (pts) => {
  const abs = Math.abs(pts).toFixed(1)
  const sign = pts < 0 ? '−' : '+'
  return `${sign}${abs} ${abs === '1.0' ? 'pt' : 'pts'}`
}

export const MOVE_FLOOR = 0.05

export const MOVE_FRACTION = 0.05

export const markMovedMaterially = (promoted, fresh) =>
  Math.abs(fresh - promoted) > Math.max(MOVE_FLOOR, promoted * MOVE_FRACTION) + 1e-9

export function requote(base, mark) {
  const half = (base.ask - base.bid) / 2
  return { bid: mark - half, ask: mark + half, mid: mark, delta: base.delta, iv: base.iv }
}

export function xlfLeg(key, { fetch, freshMark, fill, contract, promoted = true } = {}) {
  // fetch: 'fresh' | 'close' | 'unavailable' | 'missing'
  const base = contract ?? XLF[key]
  const at =
    fetch === 'close' ? XLF.closeAt : fetch === 'fresh' ? XLF.freshAt : XLF.quotedAt
  const quote =
    fetch === 'unavailable' || fetch === 'missing'
      ? null
      : { ...requote(base, freshMark ?? base.mark), at: `${longDate(XLF.today)} · ${at} ET`, time: `${at} ET` }
  return {
    key,
    ticker: XLF.ticker,
    source: 'chain',
    promoted,
    promotedMark: promoted ? base.mark : null,
    strike: base.strike,
    expiration: base.expiration,
    quote,
    noQuote: fetch === 'missing' ? 'No quote available' : null,
    fill: fill ?? (promoted ? fix2(base.mark) : ''),
    fees: '0.00',
    fillDate: XLF.today,
    errors: {},
    pairError: null,
    notice: fetch === 'unavailable' ? NOTICE.unavailable : fetch === 'close' ? NOTICE.stale : null,
    hints: promoted ? { fill: 'editable — override with your actual fill' } : {}
  }
}

export function summarize(long, short, contracts) {
  const n = num(contracts)
  const lf = num(long.fill)
  const sf = num(short.fill)
  const fees = (num(long.fees) ?? 0) + (num(short.fees) ?? 0)
  const usable = n != null && n > 0
  const cost = usable && lf != null ? lf * 100 * n : null
  const credit = usable && sf != null ? sf * 100 * n : null
  const debit = cost != null && credit != null ? cost - credit + fees : null
  const width = long.strike != null && short.strike != null ? short.strike - long.strike : null
  const ratio = lf != null && sf != null && width != null && width > 0 ? (lf - sf) / width : null
  return { cost, credit, fees, debit, width, ratio }
}

export function promoteBanner(p, long, short) {
  if (!p) return null
  const legs = { long, short }
  const live = ['long', 'short'].filter((k) => legs[k].promoted)
  if (p.fetch === 'failed') {
    return { kind: 'offline', tone: 'warning', text: `Couldn’t refresh quotes — showing screener snapshot from ${XLF.quotedAt}. Verify before recording.` }
  }
  const missing = live.find((k) => legs[k].noQuote)
  if (missing) {
    const leg = legs[missing]
    return {
      kind: 'missing',
      tone: 'warning',
      text: `The ${money(leg.strike)} ${LEG_META[missing].name} expiring ${longDate(leg.expiration)} is no longer in ${XLF.ticker}’s chain — pre-filled from the ${XLF.quotedAt} screener snapshot. Choose a contract or enter your fill manually.`
    }
  }
  if (p.session === 'CLOSED') {
    return { kind: 'stale', tone: 'warning', text: `Market closed — the pre-filled marks are stale after-hours snapshots (quoted ${XLF.closeAt}). Verify before recording.` }
  }
  if (p.session === 'EXT') {
    return { kind: 'stale', tone: 'warning', text: `Extended hours — options aren’t trading, so the pre-filled marks are stale snapshots (quoted ${XLF.closeAt}). Verify before recording.` }
  }
  const moved = live.filter((k) => legs[k].quote && markMovedMaterially(legs[k].promotedMark, legs[k].quote.mid))
  if (moved.length) {
    const freshDebit = long.quote.mid - short.quote.mid
    const width = short.strike - long.strike
    const list = moved.map((k) => `${LEG_META[k].name} quoted ${money(legs[k].promotedMark)} → now ${money(legs[k].quote.mid)}`).join(' · ')
    return { kind: 'moved', tone: 'warning', text: `Price moved: ${list} — net debit at fresh marks ${money(freshDebit)} (${pct2(freshDebit / width)} of width). Review before submitting.` }
  }
  const edited = live.filter((k) => num(legs[k].fill) != null && num(legs[k].fill) !== legs[k].promotedMark)
  if (edited.length) {
    const list = edited.map((k) => `${LEG_META[k].name} ${money(num(legs[k].fill))} vs ${money(legs[k].promotedMark)}`).join(' · ')
    return { kind: 'edited', tone: 'success', text: `Recording your entered prices, not the screener snapshot (${list}).` }
  }
  const fresh = live.filter((k) => legs[k].quote)
  if (fresh.length === live.length && fresh.length > 0) {
    const list = fresh.map((k) => `${LEG_META[k].name} ${money(legs[k].quote.mid)}`).join(' · ')
    return fresh.length === 2
      ? { kind: 'match', tone: 'success', text: `Fresh quotes ${list} — no material move from the promoted marks.` }
      : { kind: 'match', tone: 'success', text: `Fresh quote ${list} — no material move from the promoted mark.` }
  }
  return null
}

export function buildView(preset) {
  const fx = preset.fixture
  const [long, short] = preset.legs ? preset.legs() : [emptyLeg('long', fx.today), emptyLeg('short', fx.today)]
  const bothSelected = Boolean(long.source && short.source)
  const iv = bothSelected
    ? readTermStructure({
        shortIv: short.quote ? short.quote.iv : null,
        leapsIv: long.quote ? long.quote.iv : null,
        shortSource: short.source,
        leapsSource: long.source,
        quoteState: preset.quoteState
      })
    : null
  return {
    long,
    short,
    iv,
    summary: summarize(long, short, preset.contracts),
    banner: promoteBanner(preset.promoted, long, short),
    pending: preset.save === 'pending'
  }
}

      <IvTermStructureBlock iv={view.iv} long={view.long} short={view.short} />
      <CashFlows summary={view.summary} error={preset.summaryError} />
      <div>
        <button
          type="button"
          aria-expanded={advancedOpen}
          onClick={() => setAdvancedOpen((v) => !v)}
          className="flex cursor-pointer items-center gap-[8px] border-none bg-transparent px-0 py-1 font-wb-mono text-xs uppercase tracking-[0.06em] text-wb-text-muted"
        >
          <span>{advancedOpen ? '▼' : '▶'}</span>
          Advanced
        </button>
        {advancedOpen ? (
          <div className="mt-4 flex flex-col gap-5">
            <Field label="Thesis" htmlFor="pmcc-thesis" hint={promoted ? 'seeded from the watchlist note' : undefined}>
              <NumberInput id="pmcc-thesis" aria-label="Thesis" readOnly value={promoted ? fx.note : ''} placeholder="Why this trade?" />
            </Field>
            <Field label="Notes" htmlFor="pmcc-notes">
              <NumberInput id="pmcc-notes" aria-label="Notes" readOnly value="" placeholder="Additional notes…" />
            </Field>
          </div>
        ) : null}
      </div>
    </>
  )
}

export function PresetBar({ presetKey, onPick }) {
  const groups = PRESETS.reduce((acc, p) => {
    const last = acc[acc.length - 1]
    if (last && last.group === p.group) last.items.push(p)
    else acc.push({ group: p.group, items: [p] })
    return acc
  }, [])
  return (
    <div className="flex flex-col gap-2">
      {groups.map((g) => (
        <div key={g.group} className="flex flex-wrap items-center gap-1.5">
          <span className="w-[210px] font-wb-mono text-[0.62rem] uppercase tracking-[0.12em] text-wb-text-muted">{g.group}</span>
          {g.items.map((p) => (
            <button
              key={p.key}
              type="button"
              data-preset={p.key}
              onClick={() => onPick(p.key)}
              className={`cursor-pointer rounded border px-2.5 py-1 font-wb-mono text-[0.7rem] ${presetKey === p.key ? 'border-wb-gold bg-wb-gold-dim text-wb-gold' : 'border-wb-border bg-transparent text-wb-text-secondary'}`}
            >
              {p.label}
            </button>
          ))}
        </div>
      ))}
    </div>
  )
}
