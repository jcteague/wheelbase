# US-98: IVR Freshness Ring

_As a wheel trader, I want an IV rank to show how old it is, and to stop counting as met once it is too old, so a stale-rich reading never sells me premium that is not there._

> **Surfaces covered:** the treatment has no screen of its own — it lands in the **IVR cell** of the US-66 screener results row and the US-96 watchlist row, and in the watchlist **Signal** chip. Both surfaces are shown together so one verdict is visibly shared. The age is carried by a small **ring** beside the number. The ring is full when the reading is fresh and empties a quarter at a time as the reading ages; hovering it opens a tooltip that says exactly what the state means. The KO tooltip is pinned open in every state so the copy can be reviewed without hovering — hover any other ring to see the live behaviour.

## Acceptance Criteria States

- **fresh** — 0 trading days old. Full green ring. `58` renders as today. Signal `Entry ready`.
- **aging** — 2 trading days old. Ring three-quarters full. `58` keeps its color and still meets `IVR ≥ 40`.
- **stale** — 6 trading days old. Ring half full, muted number. Cannot satisfy `IVR ≥ 40`; Signal `IV too old`. KO still ranks in the screener.
- **expired** — 12 trading days old. The cell reads `exp` beside a quarter ring. Signal `IV unknown`. `exp` with a quarter ring means a reading existed and aged out (a collection-health hint); MSFT's `n/a` with no ring means nothing was ever collected. Both are unusable and behave identically in the engine.
- **earnings** — 1 trading day old but KO reported between observation and now. The ring is drawn empty in gold: unusable regardless of age. Signal `IV too old`.
- **floor** — IV-rank floor of 50 active. Stale KO `22` is not excluded; fresh AAPL `44` is.

Ring fill is a step function of the tier, not of the raw day count — a two-day-old and a three-day-old reading look the same, because they are treated the same. Colors, fonts, and chrome use the app's `--wb-*` tokens; the tooltip mirrors the shadcn `Tooltip` primitive (`src/renderer/src/components/ui/tooltip.tsx`).

export const DOW = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat']

export function dow(day) {
  // September 2026: the 1st is a Tuesday.
  return (day + 1) % 7
}

export function isTradingDay(day) {
  // Weekends plus Labor Day (Sep 7) are the only non-sessions in the window.
  const d = dow(day)
  return d !== 0 && d !== 6 && day !== 7
}

export function ageOf(s) {
  // Session closes strictly after the observed session and at or before now.
  // Now is mid-session, so today's close is not counted.
  let n = 0
  for (let d = s.obs + 1; d < s.now; d++) if (isTradingDay(d)) n += 1
  return n
}

export function tierOf(age, predates) {
  if (predates) return 'earnings'
  if (age <= 1) return 'fresh'
  if (age <= 3) return 'aging'
  if (age <= 10) return 'stale'
  return 'expired'
}

export function readingOf(s) {
  const age = ageOf(s)
  const predates = s.er != null && s.er > s.obs && s.er <= s.now
  const tier = tierOf(age, predates)
  return {
    value: s.value,
    age,
    tier,
    usable: tier === 'fresh' || tier === 'aging',
    observed: DOW[dow(s.obs)] + ' Sep ' + s.obs,
    erLabel: predates ? DOW[dow(s.er)] + ' Sep ' + s.er : null
  }
}

export const FILL = { fresh: 1, aging: 0.75, stale: 0.5, expired: 0.25, earnings: 0 }

export function ringColor(tier) {
  if (tier === 'fresh') return T.green
  if (tier === 'aging') return T.text2
  if (tier === 'earnings') return T.gold
  return T.muted
}

export function Ring({ tier, size }) {
  const px = size ?? 14
  const stroke = 2.25
  const r = (px - stroke) / 2
  const c = 2 * Math.PI * r
  const frac = FILL[tier]
  const color = ringColor(tier)
  return (
    <svg
      width={px}
      height={px}
      viewBox={`0 0 ${px} ${px}`}
      style={{ display: 'block', flexShrink: 0 }}
      aria-hidden="true"
    >
      <circle
        cx={px / 2}
        cy={px / 2}
        r={r}
        fill="none"
        stroke={tier === 'earnings' ? T.gold : T.border}
        strokeWidth={stroke}
        strokeOpacity={tier === 'earnings' ? 0.45 : 1}
      />
      {frac > 0 ? (
        <circle
          cx={px / 2}
          cy={px / 2}
          r={r}
          fill="none"
          stroke={color}
          strokeWidth={stroke}
          strokeLinecap="butt"
          strokeDasharray={`${frac * c} ${c}`}
          transform={`rotate(-90 ${px / 2} ${px / 2})`}
        />
      ) : null}
      {tier === 'earnings' ? <circle cx={px / 2} cy={px / 2} r={1.6} fill={T.gold} /> : null}
    </svg>
  )
}

export function tooltipCopy(ticker, reading) {
  const days = reading.age === 1 ? '1 trading day' : reading.age + ' trading days'
  if (reading.tier === 'earnings') {
    return {
      title: 'Predates earnings',
      body: `Observed ${reading.observed} at the close. ${ticker} reported earnings ${reading.erLabel}, after that reading. IV re-prices through a print, so this reading is unusable regardless of age — it cannot satisfy an IV condition or the IV-rank floor.`
    }
  }
  if (reading.tier === 'expired') {
    return {
      title: 'Expired',
      body: `Last reading ${days} ago (${reading.observed}). Treated as no reading: shows exp, never decides a condition, never applies the IV-rank floor. Age this old means collection has been failing for ${ticker}.`
    }
  }
  if (reading.tier === 'stale') {
    return {
      title: 'Stale',
      body: `IV rank ${reading.value}, observed ${reading.observed} — ${days} old. Too old to trust: shown for context but cannot satisfy an IV condition or the IV-rank floor. ${ticker} still ranks on yield-per-delta.`
    }
  }
  if (reading.tier === 'aging') {
    return {
      title: 'Aging',
      body: `IV rank ${reading.value}, observed ${reading.observed} — ${days} old. Still decision-usable and still counts toward conditions and the floor; expect it to have drifted a few points.`
    }
  }
  return {
    title: 'Fresh',
    body: `IV rank ${reading.value}, observed ${reading.observed} at the close — ${days} old. Current as of the last session close.`
  }
}

export function IvrCell({ ticker, reading, baseColor, size, pinned }) {
  // The shared cell. baseColor is the surface's own color for a usable value:
  // T.text on the screener, the threshold color on the watchlist.
  const [hover, setHover] = useState(false)
  const numeral = {
    fontFamily: T.mono,
    fontSize: size ?? '0.8125rem',
    fontWeight: 600,
    lineHeight: 1
  }
  if (reading == null) {
    return (
      <div
        style={{ ...numeral, color: T.muted, fontWeight: 400, textAlign: 'right' }}
        title="No IV rank collected"
      >
        n/a
      </div>
    )
  }
  const expired = reading.tier === 'expired'
  const color = reading.usable ? baseColor : T.muted
  const tip = tooltipCopy(ticker, reading)
  const open = pinned || hover
  return (
    <div
      onMouseEnter={() => setHover(true)}
      onMouseLeave={() => setHover(false)}
      style={{
        position: 'relative',
        display: 'inline-flex',
        alignItems: 'center',
        justifyContent: 'flex-end',
        gap: 7,
        width: '100%',
        cursor: 'default'
      }}
    >
      <span style={{ ...numeral, color, fontWeight: expired ? 400 : 600 }}>
        {expired ? 'exp' : reading.value}
      </span>
      <Ring tier={reading.tier} />
      {open ? (
        <Tooltip title={tip.title} body={tip.body} tier={reading.tier} pinned={pinned && !hover} />
      ) : null}
    </div>
  )
}

export function watchSignal(row, reading) {
  if (row.ticker === 'AAPL') return { label: '4.9% away', tone: 'watching', secondary: 'IV low' }
  if (reading.tier === 'expired')
    return { label: 'IV unknown', tone: 'none', secondary: 'no usable reading' }
  if (reading.tier === 'earnings')
    return {
      label: 'IV too old',
      tone: 'watching',
      secondary: 'predates ER ' + reading.erLabel.slice(4)
    }
  if (reading.tier === 'stale')

  if (reading.value >= row.ivrTrigger)
    return { label: 'Entry ready', tone: 'ready', secondary: null }
