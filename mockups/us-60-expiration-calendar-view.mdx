import { useState } from 'react'

# US-60: Expiration Calendar View

_As a wheel trader planning my week, I want a color-coded expiration calendar so I can see when management work is clustering._

## Acceptance Criteria States

- **month**: populated month view with expirations rendered directly on date cells using phase colors
- **selected**: a clicked date reveals a detail panel showing ticker, phase, strike, and DTE
- **empty**: clean month state when there are no expirations in view

---

export function Chip({ ticker, color }) {
  return (
    <div
      style={{
        padding: '3px 6px',
        borderRadius: 8,
        background: `${color}20`,
        color,
        fontSize: '0.67rem',
        fontWeight: 700
      }}
    >
      {ticker}
    </div>
  )
}
