# Epic 09: Watchlist Bench, PMCC Lens — Final Layout

_The **Watchlist** page as it looks once US-110 and US-122 have shipped. The bench is the shipped US-96 bench with one new header control: **Wheel | PMCC** switches the criteria strip, the criteria sheet, both card sections, the cards' second line and the detail panel's matching-contract card. Entries, theses, entry conditions, the three gates and the IV-reading treatment are shared and unchanged. The `wheel lens` preset is rendered by the shipped `BenchGrid` itself, so the reader can compare the two lenses side by side._

**Fictional sample data as of Friday, September 18, 2026 (LIVE, quoted 10:42:15 ET).** Last completed session Thursday, September 17. Short calls expire **Oct 23, 2026 (35 DTE)**; LEAPS expire **Sep 17, 2027 (364 DTE)** or **Jan 21, 2028 (490 DTE)**. Every figure on a diagonal is folded in this file from the two legs' marks exactly as `scorePmccPair` in `plans/us-110/data-model.md` does it: net debit = LEAPS mark − short mark; ratio = debit ÷ strike width (rounded up at 2 dp, as the engine's reason strings are); cycle yield = short mark ÷ net debit; annualized = cycle × 365 ÷ 35; score = annualized ÷ short Δ; extrinsic = LEAPS mark − (price − LEAPS strike); recovered per cycle = short mark ÷ extrinsic. DTE is counted from the valuation date; day change from each quote's previous close.

export function entry(ticker, notes, opts) {
  return {
    ticker,
    notes,
    ownBelowPrice: null,
    ivrTrigger: null,
    postEarningsOnly: false,
    coreHolding: false,
    addedAt: '2026-08-03T14:00:00Z',
    ...opts
  }
}

export function earningsOn(date) {
  // watchlist-signal.ts: daysUntil from the valuation date; within the 7-day window is a caution.
  const daysUntil = Math.round((ms(date) - ms(VALUATION)) / DAY)
  if (daysUntil < 0) return { kind: 'unknown' }
  return { kind: 'date', date, daysUntil, withinWindow: daysUntil <= 7 }
}

export const NO_EARNINGS = { kind: 'unknown' }

export function quote(price, prevClose) {
  return { price: d2(price), prevClose: prevClose === null ? null : d2(prevClose), timestamp: QUOTED_AT }
}

export function leg(strike, expiration, mark, spreadAbs, delta, openInterest) {
  // IpcPmccLeg from plans/us-110/contracts/screener-pmcc.md; bid/ask straddle the mark.
  return {
    contractId: `${expiration.replace(/-/g, '')}C${strike}`,
    strike: d4(strike),
    expiration,
    dte: dte(expiration),
    bid: d2(mark - spreadAbs / 2),
    ask: d2(mark + spreadAbs / 2),
    mark: d2(mark),
    spreadAbsolute: d2(spreadAbs),
    spreadPercent: d2((spreadAbs / mark) * 100),
    delta: d4(delta),
    openInterest,
    volume: null,
    timestamp: QUOTED_AT
  }
}

export function put(ticker, strike, mark, spreadAbs, delta, openInterest, ivRank) {
  // ScreenerCandidate: the wheel engine's arithmetic, stored as decimal strings.
  const periodYield = mark / strike
  const annualizedYield = (periodYield * 365) / dte(SHORT_EXP)
  return {
    ticker,
    contractId: `${SHORT_EXP.replace(/-/g, '')}P${strike}`,
    strike: d4(strike),
    expiration: SHORT_EXP,
    dte: dte(SHORT_EXP),
    bid: d2(mark - spreadAbs / 2),
    ask: d2(mark + spreadAbs / 2),
    mark: d2(mark),
    spreadAbsolute: d2(spreadAbs),
    spreadPercent: d2((spreadAbs / mark) * 100),
    delta: d4(delta),
    openInterest,
    volume: null,
    ivRank,
    capitalSecured: d2(strike * 100),
    periodYield: d4(periodYield),
    annualizedYield: d4(annualizedYield),
    yieldPerDelta: d4(annualizedYield / delta),
    earnings: CLEAR,
    timestamp: QUOTED_AT
  }
}

export function ratioUp(fraction) {
  // The engine's formatPercent rounds UP at 2dp (93.8333 → 93.84), so a reason string and the
  // Debit / width cell can never disagree with each other.
  return Math.ceil(fraction * 10000 - 1e-9) / 100
}

export function diagonal(ticker, long, short, price, ivRank) {
  // IpcPmccScoredCandidate, mirroring scorePmccPair in plans/us-110/data-model.md.
  const longMark = Number(long.mark)
  const shortMark = Number(short.mark)
  const netDebit = longMark - shortMark
  const width = Number(short.strike) - Number(long.strike)
  const cycle = shortMark / netDebit
  const annual = (cycle * 365) / short.dte
  const extrinsic =
    price === null ? null : Math.max(0, longMark - Math.max(0, price - Number(long.strike)))
  const coverage = extrinsic === null || extrinsic <= 0 ? null : shortMark / extrinsic
  return {
    ticker,
    long,
    short,
    ivRank,
    netDebit: d2(netDebit),
    capitalAtRisk: d2(netDebit * 100),
    strikeWidth: d2(width),
    debitToWidthPercent: d2(ratioUp(netDebit / width)),
    cycleYield: d4(cycle),
    annualizedYield: d4(annual),
    returnPerDelta: d4(annual / Number(short.delta)),
    longExtrinsic: extrinsic === null ? null : d2(extrinsic),
    extrinsicCoverage: coverage === null ? null : d4(coverage),
    earnings: CLEAR,
    timestamp: QUOTED_AT
  }
}

export function snapshotFor(mode) {
  // The watchlist half of the bench. An outage or a single failed quote nulls the price.

export const OUTAGE = { status: 'provider_unavailable', ranked: [], excluded: [], quoteTimestamp: null }

export function wheelResults(mode) {
  if (mode === 'outage') return OUTAGE
  const ranked = STOCKS.filter((s) => s.put !== null)
    .map((s) => put(s.entry.ticker, ...s.put, s.ivRank))
    .sort((a, b) => Number(b.yieldPerDelta) - Number(a.yieldPerDelta))
  const excluded = STOCKS.filter((s) => s.put === null).map((s) => ({
    ticker: s.entry.ticker,
    code: 'delta_band',
    reason: s.putExclusion
  }))
  return { status: 'ok', ranked, excluded, quoteTimestamp: QUOTED_AT }
}

export function pmccResults(mode, criteria) {
  // The funnel from plans/us-110/data-model.md: ticker (IV-rank band) → leg → pair, ranked by
  // annualized cycle yield on net debit per unit of short delta.
  if (mode === 'outage') return OUTAGE
  const ranked = []
  const excluded = []
  STOCKS.forEach((s) => {
    const ticker = s.entry.ticker
    const usable = s.ivRank !== null && (s.ivRank.state === 'fresh' || s.ivRank.state === 'aging')
    const observed = s.ivRank === null ? '' : easternDay.format(new Date(s.ivRank.observedAt))
    if (usable && criteria.ivFloor !== null && Number(s.ivRank.value) < criteria.ivFloor) {
      excluded.push({
        ticker,
        code: 'iv_rank_floor',
        reason: `IV rank ${s.ivRank.value} (${observed}) below ${criteria.ivFloor}`
      })
      return
    }
    if (usable && criteria.ivCeiling !== null && Number(s.ivRank.value) > criteria.ivCeiling) {
      excluded.push({
        ticker,
        code: 'iv_rank_ceiling',
        reason: `IV rank ${s.ivRank.value} (${observed}) above ${criteria.ivCeiling}`
      })
      return
    }
    if (s.long === null) {
      excluded.push({ ticker, code: 'no_options_listed', reason: s.diagonalExclusion })
      return
    }
    const price = mode === 'noquote' && ticker === 'XLF' ? null : Number(s.quote.price)
    const pair = diagonal(ticker, s.long, s.short, price, s.ivRank)
    if (Number(pair.debitToWidthPercent) > criteria.maxRatio) {
      excluded.push({
        ticker,
        code: 'debit_to_width',
        reason: `debit ${fmtRatioPercent(pair.debitToWidthPercent)} of width exceeds ${criteria.maxRatio}%`
      })
      return
    }
    ranked.push(pair)
  })
  ranked.sort(
    (a, b) =>
      Number(b.returnPerDelta) - Number(a.returnPerDelta) || a.ticker.localeCompare(b.ticker)
  )
  return { status: 'ok', ranked, excluded, quoteTimestamp: QUOTED_AT }
}

export function NumericInput({ ariaLabel, value, widthClass, hasError, disabled, suffix }) {
  return (
    <span
      className={[
        'inline-flex items-center gap-1.5 rounded-md border px-2.5 py-1.5',
        widthClass,
        hasError ? 'border-wb-red' : 'border-wb-border',
        disabled ? 'bg-wb-bg-surface opacity-50' : 'bg-wb-bg-base'
      ].join(' ')}
    >
      <input
        type="text"
        aria-label={ariaLabel}
        readOnly
        value={value}
        disabled={disabled}
        className="w-full min-w-0 flex-1 border-none bg-transparent font-wb-mono text-[0.85rem] text-wb-text-primary outline-none"
      />
      {suffix && <span className="font-wb-mono text-[0.78rem] text-wb-text-muted">{suffix}</span>}
    </span>
  )
}

export function RangeField({ label, caption, unit, error, minLabel, maxLabel, min, max }) {

  const [state, setState] = useState({ ...BASE_STATE, ...PRESETS[0] })
  const patch = (changes) => setState((s) => ({ ...s, ...changes }))
  const pick = (preset) => {
    setPresetKey(preset.key)
    setState({ ...BASE_STATE, ...preset })
  }

  const pmcc = state.strategy === 'PMCC'
  const criteria = { maxRatio: state.maxRatio, ivFloor: state.ivFloor, ivCeiling: state.ivCeiling }
  const snapshot = snapshotFor(state.mode)
  const results = pmcc ? pmccResults(state.mode, criteria) : wheelResults(state.mode)
  const bench = rankWithinMeets(buildBench(snapshot, results))
  const chips = pmcc ? pmccChips(criteria) : fmtCriteriaSummary(WHEEL_CRITERIA)
  const screened = results.status === 'ok'
  const stocks = [...bench.meets, ...bench.waiting]
  const current =
    stocks.find((s) => s.ticker === state.selected) ??
    stocks.find((s) => s.ticker === defaultSelection(bench)) ??
    null
  const reviewedCandidate = state.reviewed && current?.candidate ? current.candidate : null

  const openSheet = () => pmcc && patch({ sheet: 'default', saved: false })
  const saveSheet = () => {
    const band = state.sheet === 'ivband'
    patch({
      sheet: null,
      saved: true,
      ivFloor: band ? 30 : state.ivFloor,
      ivCeiling: band ? 55 : state.ivCeiling
    })
  }

  return (
    <div className="my-6 flex flex-col gap-4 text-wb-text-primary">
      <div className="flex flex-wrap items-center gap-2 text-xs text-wb-text-muted">
        <span className="mr-2 font-wb-mono uppercase tracking-widest">Preset</span>
        {PRESETS.map((preset) => (
          <Button
            key={preset.key}
            size="sm"
            variant={presetKey === preset.key ? 'secondary' : 'ghost'}
            aria-pressed={presetKey === preset.key}
            onClick={() => pick(preset)}
          >
            {preset.key}
          </Button>
        ))}
      </div>
      <div
        className={
          'relative isolate flex transform-gpu flex-col overflow-hidden rounded-xl border border-wb-border bg-wb-bg-base ' +
          (state.sheet !== null ? 'h-[1040px]' : '')
        }
      >
        <BenchHeader
          strategy={state.strategy}
          onStrategy={(strategy) => patch({ strategy, saved: false, reviewed: false, sheet: null })}
          chips={chips}
          onOpenCriteria={openSheet}
        />
        <div className="flex-1 overflow-auto">
          <div className="flex flex-col gap-5 p-6">
            {state.saved && <SavedBanner />}
            {results.status === 'provider_unavailable' && (
              <MarketDataOutage marketDataConfigured onRetry={() => {}} onOpenSettings={() => {}} />
            )}
            {pmcc ? (
              <PmccBenchGrid
                bench={bench}
                selected={state.selected}

                onReview={() => patch({ reviewed: true })}
                onAdjustCriteria={openSheet}
