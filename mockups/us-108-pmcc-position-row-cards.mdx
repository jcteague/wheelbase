# US-108: PMCC position card on the dashboard

_A PMCC is a position with a **legs array**, not a row with stacked cells. One calm position line answers "does either leg need me this week?" — glyphs, net P&L, running cost basis. Under it, one block per leg. Three contracts, a second short call or partial coverage add blocks rather than new cell types, so the same primitive carries conventional wheels when multi-contract lands. Tier 2 opens from the Detail bar at the card foot (a click anywhere on the card also toggles it); the ticker and the Open ↗ link go to the detail page, where tier 3 lives._

## Acceptance Criteria States

- **both-legs** — Background. `+$400.00` / `+18.9% of basis`, short call `50.0% of max` with the gold TARGET badge
- **falling-stock** — `-$500.00` / `-23.6% of basis` with a short call at `88.9% of max`; no position-level % of max
- **wide-leaps-spread** — LEAPS spread `$3.00` = 11.5% of mid; the ⚠ travels from the LEAPS mark onto the P&L
- **missing-leaps** — no LEAPS snapshot: mark and P&L read `—`, the short-call cue and both cash figures survive
- **short-expiring** — short call 4 DTE with earnings in 2 days: gold DTE, ⚑ on US-56's predicate, and a gold ▲ because $106.10 is 1.8% below the strike (US-106)
- **short-itm** — short through the strike: red ▲ on US-106's in-the-money branch; the `$0.08` extrinsic left shows in tier 2
- **below-breakeven** — short strike under LEAPS strike + basis: ✕, being called away locks in a loss
- **leaps-only** — short expired worthless, none sold: `+$240.00` / `+10.4%`, short term drops out
- **roll-window** — LEAPS `59d` gold at the US-107 threshold; short `31d` unemphasised, no expiring-soon flag
- **leaps-delta-low** — LEAPS delta `0.54`: δ glyph, no longer tracking as synthetic stock
- **leaps-paid-for** — basis `-$40.00` / "LEAPS paid for" in gold; `+$3,285.00` with no percentage
- **three-contracts** — same per-share fills and quotes ×3: totals scale, per-share figures do not
- **two-shorts** _(not yet supported)_ — two short calls against one LEAPS stack as two leg blocks
- **partial-coverage** _(not yet supported)_ — 3 LEAPS, 1 short sold: the coverage ratio is explicit
- **closed** — muted, in the Closed section, last basis and credits only

Every figure is derived from the preset's fills, quotes, contracts and valuation date, so this mockup and the
acceptance criteria cannot drift apart. The derivation panel under the grid prints the arithmetic.

export const d = (s) => new Date(s + 'T00:00:00Z')
export const dte = (from, to) => Math.round((d(to) - d(from)) / 86400000)
export const addDays = (from, n) => new Date(d(from).getTime() + n * 86400000).toISOString().slice(0, 10)
export const r2 = (n) => Math.round(n * 100) / 100

export function fmtMoney(n, signed) {
  if (n == null) return '—'
  const sign = signed ? (n > 0 ? '+' : n < 0 ? '-' : '') : n < 0 ? '-' : ''
  return sign + '$' + Math.abs(n).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })
}
export const fmtPct = (n, signed) => (signed && n > 0 ? '+' : n < 0 ? '-' : '') + Math.abs(n).toFixed(1) + '%'
// A subtraction term that stays legible when the operand is negative.
export const term = (n) => (n < 0 ? '+ ' + fmtMoney(Math.abs(n)) : '− ' + fmtMoney(n))
export function mark(q) {
  if (!q || q.bid == null) return { mid: null, unavailable: true }
  const mid = r2((q.bid + q.ask) / 2)
  const spread = r2(q.ask - q.bid)
  const spreadPct = (spread / mid) * 100
  return { mid, spread, spreadPct, wide: spreadPct > WIDE_SPREAD_PCT, unavailable: false }
}
// Pure valuation — mirrors computePmccUnrealizedPnl in src/main/core/costbasis.ts,
// generalised over a legs array so N shorts need no second formula.
export function computePmccUnrealizedPnl({ legs, basisTotal }) {
  const gross = legs.reduce((s, l) => s + (l.side === 'SHORT' ? -1 : 1) * l.mid * 100 * l.contracts, 0)
  const pnl = r2(gross - basisTotal)
  return { pnl, pnlPercentOfBasis: basisTotal > 0 ? (pnl / basisTotal) * 100 : null, gross: r2(gross) }
}

export function derive(p) {
  const closed = p.status === 'CLOSED'
  const legs = p.legs.map((l) => ({ ...l, m: mark(l.q), dte: dte(p.valuation, l.exp) }))
  const longs = legs.filter((l) => l.side === 'LONG')
  const shorts = legs.filter((l) => l.side === 'SHORT')
  const longCt = longs.reduce((s, l) => s + l.contracts, 0)
  const shortCt = shorts.reduce((s, l) => s + l.contracts, 0)
  const basisTotal = r2(p.basisPerShare * 100 * longCt)
  const premiumTotal = r2(p.creditsToDate * 100 * longCt)
  const recovered = (p.creditsToDate / p.leapsCostToDate) * 100
  const breakeven = r2(longs[0].strike + p.basisPerShare)
  const marksReady = !closed && legs.every((l) => l.m.mid != null)
  const val = marksReady ? computePmccUnrealizedPnl({
    legs: legs.map((l) => ({ side: l.side, mid: l.m.mid, contracts: l.contracts })), basisTotal }) : null
  // Short-call cues reuse computeUnrealizedPnl / resolveProfitTarget, fed the short leg.
  legs.forEach((l) => {
    if (l.side !== 'SHORT' || closed || l.m.mid == null) return
    l.pnl = r2((l.fill - l.m.mid) * 100 * l.contracts)
    l.pctMax = ((l.fill - l.m.mid) / l.fill) * 100
    l.target = l.pctMax >= p.target * 100
    l.itm = p.price >= l.strike
    l.extrinsic = r2(Math.max(0, l.m.mid - Math.max(0, p.price - l.strike)))
  })
  const nearShort = shorts.slice().sort((a, b) => a.dte - b.dte)[0] || null
  const wideLeaps = longs.some((l) => l.m.wide)
  const alerts = []
  if (!closed) {
    if (longs.some((l) => l.dte <= ROLL_WINDOW_DTE)) alerts.push({ g: '◆', n: 'Roll window', c: T.gold, t: `LEAPS inside the ${ROLL_WINDOW_DTE}-day roll window (US-107)` })
    // ▲ is US-106's one-directional band: within 2% below the short strike, or anywhere at/above it.
    const pctBelow = (l) => ((l.strike - p.price) / l.strike) * 100
    if (shorts.some((l) => l.itm)) alerts.push({ g: '▲', n: 'Assignment risk', c: T.red, t: `Stock is ${Math.abs(pctBelow(shorts.find((l) => l.itm))).toFixed(1)}% above the short-call strike — in the money, LEAPS would have to cover (US-106)` })
    else if (shorts.some((l) => pctBelow(l) <= ASSIGNMENT_RISK_PCT_BELOW)) alerts.push({ g: '▲', n: 'Assignment risk', c: T.gold, t: `Stock is ${pctBelow(shorts.find((l) => pctBelow(l) <= ASSIGNMENT_RISK_PCT_BELOW)).toFixed(1)}% below the short-call strike — approaching assignment (US-106)` })
    // ⚑ is US-56's predicate: 0 ≤ days to earnings ≤ 10 and on or before the short call's expiration.
    const daysToEarnings = p.earnings ? dte(p.valuation, p.earnings) : null
    if (daysToEarnings !== null && nearShort && daysToEarnings >= 0 && daysToEarnings <= EARNINGS_PROXIMITY_MAX_DAYS && daysToEarnings <= nearShort.dte) alerts.push({ g: '⚑', n: 'Earnings', c: T.gold, t: `Earnings in ${daysToEarnings} days before your ${nearShort.exp} expiration (US-56)` })
    if (shorts.some((l) => l.strike < breakeven)) alerts.push({ g: '✕', n: 'Below breakeven', c: T.red, t: `Short strike below breakeven ${fmtMoney(breakeven)} — being called away locks in a loss` })
    if (longs.some((l) => l.delta < LEAPS_DELTA_FLOOR)) alerts.push({ g: 'δ', n: 'Delta floor', c: T.gold, t: `LEAPS delta ${longs[0].delta.toFixed(2)} below ${LEAPS_DELTA_FLOOR.toFixed(2)} — no longer tracking as synthetic stock` })
    if (wideLeaps) alerts.push({ g: '⚠', n: 'Wide spread', c: T.gold, t: `LEAPS spread is ${longs.find((l) => l.m.wide).m.spreadPct.toFixed(1)}% of the mid — the P&L that depends on it is unreliable` })
  }
  return { closed, legs, longs, shorts, longCt, shortCt, basisTotal, premiumTotal, recovered,
    breakeven, val, nearShort, wideLeaps, alerts, cycles: p.cycles,
    sortKey: nearShort ? nearShort.dte : Math.min(...longs.map((l) => l.dte)),
    partial: shortCt < longCt,
    state: closed ? 'Closed' : shorts.length ? (shorts.length > 1 ? `LEAPS + ${shorts.length} short calls open` : 'LEAPS + short call open') : 'LEAPS open · no short call' }
}

// Tier 2 — credits and cycles, breakeven, max if called, delta, extrinsic, mark width, entry debit.
export function t2cells(p, x) {
  const l0 = x.longs[0]
  const s0 = x.nearShort
  const maxIfCalled = s0 ? r2((s0.strike - l0.strike - p.basisPerShare) * 100 * s0.contracts) : null
  return [
    ['Credits collected', fmtMoney(x.premiumTotal), `${p.cycles} short-call cycles · ${fmtPct(x.recovered)} of LEAPS cost`],
    ['Breakeven', fmtMoney(x.breakeven), `LEAPS ${fmtMoney(l0.strike)} + basis ${fmtMoney(p.basisPerShare)} / share`],
    ['Max if called', s0 ? fmtMoney(maxIfCalled, true) : '—', s0 ? `called away at ${fmtMoney(s0.strike)}` : 'no short call open'],
    ['LEAPS delta', l0.delta.toFixed(2), l0.delta < LEAPS_DELTA_FLOOR ? `below the ${LEAPS_DELTA_FLOOR.toFixed(2)} floor` : 'tracking as synthetic stock'],
    ['Short extrinsic', s0 && s0.extrinsic != null ? fmtMoney(s0.extrinsic) : '—', s0 ? (s0.itm ? 'ITM — assignment math is live' : 'OTM — all extrinsic') : 'no short call open'],
    ['LEAPS mark', l0.m.mid == null ? '—' : fmtMoney(l0.m.mid),
      l0.m.mid == null ? 'no snapshot this poll' : `${fmtMoney(l0.q.bid)} × ${fmtMoney(l0.q.ask)} · ${l0.m.spreadPct.toFixed(1)}% wide${l0.m.wide ? ' · P&L unreliable' : ''}`],
    ['Entry net debit', fmtMoney(p.entryDebit * 100 * x.longCt), `${fmtMoney(p.entryDebit)} / share · opened ${p.openDate} · ${dte(p.openDate, p.valuation)}d ago`]
  ]
}

export function Derivation({ p }) {
  const x = derive(p)
  const l0 = x.longs[0]
  const expr = x.val
    ? x.legs.map((l) => `${l.side === 'SHORT' ? '− ' : ''}${fmtMoney(l.m.mid)}×100×${l.contracts}`).join(' ') + ' ' + term(x.basisTotal)
    : x.closed ? 'closed — no live marks' : 'a mark is missing on an open leg'
  const rows = [
    ['Valuation date', 'America/New_York', p.valuation],
    ['Legs', `${x.longCt} LEAPS contract${x.longCt === 1 ? '' : 's'} · ${x.shortCt} short${x.shortCt === 1 ? '' : 's'}${x.partial ? ' · part-covered' : ''}`, x.legs.length + ' blocks'],
    ['LEAPS mid', l0.q ? `(${fmtMoney(l0.q.bid)} + ${fmtMoney(l0.q.ask)}) ÷ 2` : 'no snapshot returned this poll', l0.q ? fmtMoney(l0.m.mid) : '—'],
    ['LEAPS spread', l0.q ? `${fmtMoney(l0.m.spread)} ÷ ${fmtMoney(l0.m.mid)}` : '—', l0.q ? l0.m.spreadPct.toFixed(1) + '%' + (l0.m.wide ? ' · wide' : '') : '—'],
    ['Net P&L', expr, x.val ? fmtMoney(x.val.pnl, true) : '—'],
    ['% of basis', x.val ? (x.val.pnlPercentOfBasis == null ? 'undefined — basis is not positive' : `${fmtMoney(x.val.pnl, true)} ÷ ${fmtMoney(x.basisTotal)}`) : '—',
      x.val && x.val.pnlPercentOfBasis != null ? fmtPct(x.val.pnlPercentOfBasis, true) : '—'],
    ...x.shorts.map((s, i) => [`Short ${x.shorts.length > 1 ? i + 1 + ' ' : ''}% of max`,
      s.pctMax != null ? `(${fmtMoney(s.fill)} − ${fmtMoney(s.m.mid)}) ÷ ${fmtMoney(s.fill)}` : '—',
      s.pctMax != null ? fmtPct(s.pctMax) : '—']),
    ['Breakeven', `LEAPS ${fmtMoney(l0.strike)} + basis ${fmtMoney(p.basisPerShare)}`, fmtMoney(x.breakeven)],
    ['Premium recovered', `${fmtMoney(p.creditsToDate)} credits ÷ ${fmtMoney(p.leapsCostToDate)} LEAPS cost`, fmtPct(x.recovered)],
    ['Cost basis', `${fmtMoney(p.basisPerShare)} / share × 100 × ${x.longCt}`, fmtMoney(x.basisTotal)],
    ['Sort key', x.nearShort ? 'short-call DTE (nearest open short)' : 'LEAPS DTE (no short call)', x.closed ? 'closed' : x.sortKey + 'd']
  ]
  return (
    <div className="panel">
      <div style={{ padding: '10px 14px', borderBottom: '1px solid var(--wb-border, #21262d)', fontFamily: 'var(--font-wb-mono, ui-monospace, Menlo, monospace)', fontSize: 10, fontWeight: 600, letterSpacing: '.14em', textTransform: 'uppercase', color: p.future ? T.gold : T.mut }}>
        Derivation · {p.note}
      </div>
      <div className="deriv">
        {rows.map(([k, e, v], i) => (
          <Fragment key={i}>
            <div className="dk">{k}</div><div className="de">{e}</div><div className="dv">{v}</div>
          </Fragment>
        ))}
      </div>
    </div>
  )
}
