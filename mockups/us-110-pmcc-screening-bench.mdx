# US-110: PMCC lens on the watchlist bench

_The shipped US-96 bench, with one new control. **Wheel | PMCC** in the header switches the screen from "best put per ticker" to "best LEAPS + short-call diagonal per ticker". Entries, theses, entry conditions and the three gates are shared; only the criteria, the engine and the match description change._

**Fictional sample data as of Friday, September 18, 2026, 10:42 CT.** Last completed session Thursday, September 17. Short calls expire **Oct 23, 2026 (35 DTE)**; LEAPS expire **Sep 17, 2027 (364 DTE)** or **Jan 21, 2028 (490 DTE)**. Every number on a diagonal is computed in this file from the two legs' marks: net debit = LEAPS mark − short mark; ratio = debit ÷ strike width; cycle yield = short mark ÷ debit; annualized = cycle × 365 ÷ 35; score = annualized ÷ short Δ; extrinsic = LEAPS mark − (price − LEAPS strike).

export const SHORT_DTE = 35

export function money(n) {
  return '$' + n.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })
}

export function pct(fraction) {
  // fmtYieldPercent: 2dp half-up, trailing zeros trimmed.
  return String(Number((fraction * 100).toFixed(2))) + '%'
}

export function pctUp(fraction) {
  // The engine's formatPercent rounds UP at 2dp (93.8333 → 93.84), so a reason string and the
  // Debit / width cell can never disagree with each other.
  return String(Math.ceil(fraction * 10000 - 1e-9) / 100) + '%'
}

export function diagonal(price, long, short) {
  // Mirrors scorePmccPair in plans/us-110/data-model.md, in plain numbers for the mockup.
  const netDebit = long.mark - short.mark
  const width = short.strike - long.strike
  const cycle = short.mark / netDebit
  const annual = (cycle * 365) / short.dte
  const extrinsic = price === null ? null : Math.max(0, long.mark - Math.max(0, price - long.strike))
  return {
    long,
    short,
    netDebit,
    perSpread: netDebit * 100,
    width,
    ratio: (netDebit / width) * 100,
    cycle,
    annual,
    score: annual / short.delta,
    extrinsic,
    coverage: extrinsic === null || extrinsic <= 0 ? null : short.mark / extrinsic
  }
}

export function evaluate(stock, strategy, mode, maxRatio) {
  // Mirrors lib/bench.ts: outage → trader's gates → screener exclusion → not screened.
  const pmcc = strategy === 'PMCC'
  let match = null
  let exclusion = pmcc ? stock.diagonalExclusion : stock.putExclusion
  if (pmcc && stock.long !== null) {
    const pair = diagonal(mode === 'outage' ? null : stock.price, stock.long, stock.short)
    if (pair.ratio > maxRatio) {
      exclusion = 'debit ' + pctUp(pair.ratio / 100) + ' of width exceeds ' + maxRatio + '%'
    } else {
      match = pair
    }
  } else if (!pmcc && stock.put !== null) {
    const period = stock.put.mark / stock.put.strike
    const annual = (period * 365) / SHORT_DTE
    match = { ...stock.put, period, annual, score: annual / stock.put.delta }
  }
  if (mode === 'empty') {
    exclusion = pmcc
      ? 'debit ' + pctUp(match ? match.ratio / 100 : 0.95) + ' of width exceeds 60%'
      : 'delta 0.24 outside 0.10–0.15'
    match = null
  }
  const meets = mode === 'normal' && stock.gates.length === 0 && match !== null
  const reason =
    mode === 'outage'
      ? 'Data unavailable · not evaluated'
      : stock.gates.length > 0
        ? stock.gates.join(' · ')
        : (exclusion ?? 'Not screened yet')
  const verdict =
    stock.conditions.some((c) => c.verdict === 'met') && stock.conditions.every((c) => c.verdict === 'met')
      ? 'All conditions met'
      : 'Screening criteria met'
  return { ...stock, match, meets, reason, verdictCopy: meets ? verdict : null }
}

    ' yield) but is held back by the entry conditions above.'
  )
}

export function BenchDetail({ stock, strategy }) {

  const tone =
    change === '—'

      : change.startsWith('−')

        : 'text-wb-green'

            Edit
          </Button>

      <div className="grid grid-cols-3 gap-3 rounded-md bg-wb-bg-elevated p-4">
        <div>
          <p className="m-0 text-xs text-wb-text-muted">Last price</p>

        <h4 className="mb-2 text-xs font-semibold uppercase tracking-widest text-wb-text-muted">

        </h4>
        <p className="m-0 text-sm leading-relaxed text-wb-text-secondary">{stock.thesis}</p>

      <div>
        <h4 className="mb-2 text-xs font-semibold uppercase tracking-widest text-wb-text-muted">
          Entry conditions
        </h4>
        <div className="flex flex-wrap gap-2">
          {stock.conditions.length === 0 && <Badge>No personal conditions</Badge>}
          {stock.conditions.map((c) => (
            <Badge key={c.label} color={VERDICT_COLOR[c.verdict]}>
              {c.label} · {c.verdict}
            </Badge>
          ))}
        </div>
      </div>
      <div className="flex justify-between gap-3 text-xs">
        <span className="text-wb-text-muted">Earnings</span>
        <span className={stock.gates.some((g) => g.startsWith('Earnings')) ? 'text-wb-gold' : undefined}>
          {stock.earnings}
        </span>
      </div>
      {stock.meets ? (
        strategy === 'PMCC' ? (
          <MatchingDiagonalCard match={stock.match} />
        ) : (
          <MatchingPutCard match={stock.match} />
        )
      ) : (
        <div className="flex flex-col gap-3">

            {stock.reason}. This stock stays on your watchlist while you wait.
          </AlertBox>
          {stock.match !== null && (
            <p className="m-0 text-xs leading-relaxed text-wb-text-muted">
              {heldBackLine(stock, strategy)}
            </p>
          )}
        </div>
      )}
    </div>
  )
}

  const [mode, setMode] = useState('normal')
  const [maxRatio, setMaxRatio] = useState(90)
  const [selected, setSelected] = useState('XLF')
  const [saved, setSaved] = useState(false)
  const pmcc = strategy === 'PMCC'
  const evaluated = STOCKS.map((s) => {
    const e = evaluate(s, strategy, mode, pmcc ? maxRatio : 90)
    return {
      ...e,
      priceText: mode === 'outage' ? '—' : money(s.price),
      changeText: mode === 'outage' ? '—' : s.change
    }
  })
  const meets = evaluated
    .filter((s) => s.meets)
    .sort((a, b) => b.match.score - a.match.score)
    .map((s, i) => ({ ...s, rank: i + 1 }))
  const waiting = evaluated.filter((s) => !s.meets)
  const all = [...meets, ...waiting]
  const current = all.find((s) => s.ticker === selected) ?? all[0]
  const chips = pmcc ? pmccChips(maxRatio) : WHEEL_CHIPS

    setSaved(false)

    if (name === 'excluded') setSelected('AMD')
    if (name === 'held') setSelected('AAPL')

    if (name === 'outage') setMode('outage')

      setSaved(true)
      setSelected('KO')
