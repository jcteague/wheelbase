# Epic 09: PMCC Dashboard — Final Layout

_The positions dashboard as it looks once every Epic 09 story that touches it has shipped. **XYZ** is the PMCC from the detail mockup — same LEAPS, same nine short-call cycles, same quotes per preset — sitting in one card grid with three wheels in different phases and a closed section. Every figure on every card is folded from those legs and quotes; the action strip, the awareness tags, the management queue rows and the IVR context cells are all derived from the same predicates, so the live read on the card and the managed read in the queue can be compared preset by preset._

## Signals: action states and awareness tags

A card's signals come in two tiers. **Action states** mean a leg needs a trade. On the PMCC card they are ✕ short strike below breakeven (US-108, red), ▲ assignment risk (US-106, red in the money, gold approaching), ◆ LEAPS roll window (US-107, gold), and the short-leg queue rules riding on the nearest short call, EXPIRATION_IMMINENT and MANAGEMENT_WINDOW (gold, with the queue's own summary). On a wheel card they are that card's own STRIKE_PROXIMITY (red below a put strike, gold otherwise), COVERED_CALL_BREACH (red), EXPIRATION_IMMINENT and MANAGEMENT_WINDOW rows, whose strip text is the queue summary word for word. Any open action state puts a full-width **action strip** at the top of the card. The strip shows the glyph, the status word from US-118's detail-page lanes, and the concrete figure. The leg the state belongs to takes the same tint, a tone-coloured left border and the small glyph, and the card's left edge takes the action tone in place of the strategy colour. The teal PMCC pill still names the strategy. When several states are open, the strip leads with the first in US-118's lane order, ✕ BELOW BREAKEVEN › ▲ ASSIGNMENT RISK red › ▲ ASSIGNMENT RISK gold › ◆ ROLL WINDOW › EXPIRATION_IMMINENT › MANAGEMENT_WINDOW, and a `+N` chip lists the rest in its tooltip. **Awareness tags** (⚑ earnings, ⚠ wide spread, δ delta floor) stay small after the PMCC pill, because they qualify a read rather than ask for a trade. PROFIT_TARGET is good news and keeps its TARGET chip. US-108's glyph-row wording needs to change to match: ▲, ◆ and ✕ no longer sit in a glyph row after the ticker; they are carried by the strip and the leg.

export const DAY = 86400000

export const ms = (s) => Date.parse(`${s}T00:00:00Z`)

export const days = (a, b) => Math.round((ms(b) - ms(a)) / DAY)

export const isoDate = (t) => {
  const d = new Date(t)
  return `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, '0')}-${String(d.getUTCDate()).padStart(2, '0')}`
}

export const addDays = (from, n) => isoDate(ms(from) + n * DAY)

export const money = (n) =>
  `${n < 0 ? '-' : ''}$${Math.abs(n).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`

export const signed = (n) => `${n < 0 ? '−' : '+'}${money(Math.abs(n))}`

export const pct = (x, d = 1) => `${(x * 100).toFixed(d)}%`

export const signedPct = (x, d = 1) => `${x < 0 ? '−' : '+'}${pct(Math.abs(x), d)}`

export const mid = (bid, ask) => (bid == null || ask == null ? null : (bid + ask) / 2)

export function cycleShape(cycle, today) {
  const current = cycle.rolls.length ? cycle.rolls[cycle.rolls.length - 1].open : cycle.open
  const outcome = cycle.end ? cycle.end.outcome : 'open'
  const endDate = cycle.end ? cycle.end.date : today
  const rollNet = cycle.rolls.reduce((sum, r) => sum + (r.open.price - r.close.price), 0)
  const closeCost = cycle.end && cycle.end.outcome === 'closed' ? cycle.end.price : 0
  return {
    ...cycle,
    current,
    outcome,
    endDate,
    dteAtOpen: days(cycle.open.date, cycle.open.expiration),
    daysHeld: days(cycle.open.date, endDate),
    net: cycle.open.price + rollNet - closeCost,
    rollNet,
    closeCost
  }
}

export function buildModel(preset) {
  // Copied from epic-09-pmcc-position-detail.mdx — the one ledger every Epic 09 screen folds from.
  const today = preset.today
  const contracts = preset.contracts ?? 1
  const leapsEvents = preset.leaps ?? BASE_LEAPS
  const rawCycles = preset.cycles ?? (preset.patch ? preset.patch(BASE_CYCLES) : BASE_CYCLES)
  const cycles = rawCycles.map((c) => cycleShape(c, today))
  const ledger = []
  let leapsCost = 0
  let credits = 0
  const push = (date, label, group) => {
    ledger.push({ date, label, group, basis: leapsCost - credits })
  }
  const timed = []
  leapsEvents.forEach((e) => timed.push({ at: e.date, order: 0, kind: 'leaps', e }))
  cycles.forEach((c) => {
    timed.push({ at: c.open.date, order: 3, kind: 'cycle-open', c })
    c.rolls.forEach((r) => timed.push({ at: r.date, order: 2, kind: 'cycle-roll', c, r }))
    if (c.end) timed.push({ at: c.end.date, order: 1, kind: 'cycle-end', c })
  })
  timed
    .sort((a, b) => ms(a.at) - ms(b.at) || a.order - b.order)
    .forEach((t) => {
      if (t.kind === 'leaps') {
        if (t.e.kind === 'open') leapsCost += t.e.price
        else leapsCost += t.e.open.price - t.e.close.price
        push(t.at, t.e.kind === 'open' ? 'LEAPS bought' : 'LEAPS roll #1', 'leaps')
      } else if (t.kind === 'cycle-open') {
        credits += t.c.open.price
        push(t.at, `Cycle ${t.c.n} opened`, `cycle-${t.c.n}`)
      } else if (t.kind === 'cycle-roll') {
        credits += t.r.open.price - t.r.close.price
        push(t.at, `Cycle ${t.c.n} roll`, `cycle-${t.c.n}`)
      } else {
        if (t.c.end.outcome === 'closed') credits -= t.c.end.price
        push(t.at, `Cycle ${t.c.n} ${OUTCOME_LABEL[t.c.end.outcome].toLowerCase()}`, `cycle-${t.c.n}`)
      }
    })
  const openLeaps = leapsEvents[leapsEvents.length - 1]
  const currentLeaps = openLeaps.kind === 'open' ? openLeaps : openLeaps.open
  const start = leapsEvents[0].date
  const closed = cycles.filter((c) => c.end)
  const openCycle = cycles.find((c) => !c.end)
  const realized = closed.reduce((s, c) => s + c.net, 0)
  return {
    today,
    contracts,
    start,
    leapsEvents,
    currentLeaps,
    cycles,
    ledger,
    openCycle,
    openCycles: cycles.filter((c) => !c.end),
    closedCount: closed.length,
    realized,
    leapsCost,
    credits,
    basis: leapsCost - credits,
    entryDebit: leapsEvents[0].price - (cycles[0] && cycles[0].open.date === start ? cycles[0].open.price : 0),
    positionClosed: Boolean(preset.closeAll),
    closedAt: preset.closedAt,
    imported: Boolean(preset.imported)
  }
}

export function markOf(bid, ask) {
  const m = mid(bid, ask)
  return { bid, ask, mid: m, spread: m == null ? null : ask - bid }
}

export function ivrZone(value) {
  const v = Number(value)
  return v < IVR_LOW_MAX ? 'low' : v > IVR_HIGH_MIN ? 'high' : 'mid'
}

export const tradingDaysLabel = (n) => `${n} trading ${n === 1 ? 'day' : 'days'}`

export function unusable(ivRank) {
  if (ivRank === null) return { status: 'n/a', tone: 'muted', text: 'no IV rank collected' }
  const age = tradingDaysLabel(ivRank.ageTradingDays)
  if (ivRank.state === 'stale') return { status: 'Stale · —', tone: 'muted', text: `${age} old — collect a fresh reading before acting on it` }
  if (ivRank.state === 'expired') return { status: 'Expired · —', tone: 'muted', text: `last reading ${age} ago — collection has been failing` }
  if (ivRank.state === 'predates_earnings') return { status: 'Predates earnings · —', tone: 'muted', text: 'observed before the last print — IV has re-priced since' }
  return null
}

export function resolveContext(registry, input) {
  const blocked = unusable(input.ivRank)
  if (blocked) return { ...blocked, rule: 'unusable' }
  const i = { ...input, zone: ivrZone(input.ivRank.value) }
  const idx = registry.findIndex((r) => r.test(i))
  const r = registry[idx]
  return { status: r.status, tone: r.tone, text: r.text(i), rule: `row ${idx + 1}` }
}

export function faceZone(ivRank, shortCtx) {
  // The face word is the zone; its tone is the short-call read, so an event-driven rich reading is gold at a glance.
  if (ivRank === null) return { word: null, tone: 'muted', zone: 'none' }
  if (!isUsableIvrState(ivRank.state)) return { word: TIER_TITLE[ivRank.state], tone: 'muted', zone: 'none' }
  const zone = ivrZone(ivRank.value)
  return { word: ZONE_WORD[zone], tone: shortCtx ? shortCtx.tone : ZONE_TONE[zone], zone }
}

export function pmccActions(x) {
  // The action tier, in US-118's lane order (rank): ✕ › ▲ red › ▲ gold › ◆ › EXPIRATION_IMMINENT › MANAGEMENT_WINDOW.
  // ✕ ▲ ◆ come from the same glyph evaluation the tooltips use; the short-leg queue rules ride on the nearest short (US-106) with the queue's own summary. ⚑ ⚠ δ stay awareness tags.
  const glyphActs = x.glyphs.filter((g) => PMCC_ACTION[g.glyph]).map((g) => {
    const a = PMCC_ACTION[g.glyph]
    return { glyph: g.glyph, word: a.word, tone: g.toneKey, rank: a.rank(x), reason: a.reason(x), leg: a.leg(x), owner: g.owner }
  })
  const legActs = x.closed || !x.nearShort ? [] : shortLegRows('XYZ', true, x.nearShort).filter((r) => QUEUE_ACTION[r.code]).map((r) => ({ ...queueAction(r), leg: x.nearShort, owner: 'US-106' }))
  return rankActions([...glyphActs, ...legActs])
}

export function deriveCard(preset) {
  const model = buildModel(preset)
  const q = preset.q ?? null
  // A caller's preset may carry a stock price with no option marks, or no stock price at all; marks then read — and ▲ cannot be evaluated.
  const hasStock = q != null && q.stock != null
  const closed = model.positionClosed
  const today = model.today
  const c = model.contracts
  const shortC = preset.shortContracts ?? c
  const cur = model.currentLeaps
  const leapsMark = markOf(q ? q.leapsBid : null, q ? q.leapsAsk : null)
  const leaps = {
    side: 'LONG', tag: 'L', strike: cur.strike, exp: cur.expiration, dte: days(today, cur.expiration), contracts: c,
    fill: cur.price, delta: q ? q.leapsDelta : null, m: leapsMark,
    wide: leapsMark.mid != null && leapsMark.spread > WIDE_SPREAD * leapsMark.mid
  }
  const shortQuotes = q ? (q.shorts ?? [{ bid: q.shortBid, ask: q.shortAsk }]) : []
  // A closed card still shows its last short call, dashed — "both mids and both DTEs read —" (US-108).
  const shortCycles = closed ? model.cycles.slice(-1) : model.openCycles
  const shorts = shortCycles
    .map((cy, i) => {
      const sq = shortQuotes[i] ?? { bid: null, ask: null }
      const m = markOf(sq.bid, sq.ask)
      const captured = m.mid != null ? (cy.net - m.mid) / cy.net : null
      return {
        side: 'SHORT', tag: 'S', cycle: cy.n, strike: cy.current.strike, exp: cy.current.expiration, dte: days(today, cy.current.expiration),
        contracts: shortC, fill: cy.net, m, captured,
        target: captured != null && captured * 100 >= PROFIT_TARGET,
        pnl: m.mid != null ? (cy.net - m.mid) * 100 * shortC : null,
        itm: hasStock ? q.stock >= cy.current.strike : false,
        pctBelow: hasStock ? (cy.current.strike - q.stock) / cy.current.strike : null,
        extrinsic: m.mid != null && hasStock ? Math.max(0, m.mid - Math.max(0, q.stock - cy.current.strike)) : null
      }
    })
    .sort((a, b) => a.dte - b.dte)
  const legs = [leaps, ...shorts]
  const basisTotal = model.basis * 100 * c
  const creditsTotal = model.credits * 100 * c
  const recovered = model.leapsCost > 0 ? model.credits / model.leapsCost : null
  const breakeven = cur.strike + model.basis
  const marksReady = !closed && legs.every((l) => l.m.mid != null)
  const gross = marksReady ? legs.reduce((s, l) => s + (l.side === 'SHORT' ? -1 : 1) * l.m.mid * 100 * l.contracts, 0) : null
  const pnl = marksReady ? gross - basisTotal : null
  const pnlPct = pnl != null && basisTotal > 0 ? pnl / basisTotal : null
  const nearShort = shorts[0] ?? null
  const sortKey = closed ? null : nearShort ? nearShort.dte : leaps.dte
  const daysToEarnings = preset.earnings ? days(today, preset.earnings) : null
  const earningsFlag = Boolean(!closed && nearShort && daysToEarnings != null && daysToEarnings >= 0 && daysToEarnings <= EARNINGS_PROXIMITY_MAX_DAYS && daysToEarnings <= nearShort.dte)
  const inRollWindow = !closed && leaps.dte >= 0 && leaps.dte <= ROLL_WINDOW_DTE
  const riskShort = closed ? null : (shorts.find((s) => s.pctBelow != null && s.pctBelow * 100 <= ASSIGNMENT_RISK_PCT_BELOW) ?? null)
  const x = {
    preset, model, closed, today, contracts: c, shortContracts: shortC, leaps, shorts, legs, basisTotal, creditsTotal, recovered, breakeven,
    marksReady, gross, pnl, pnlPct, nearShort, sortKey, daysToEarnings, earningsFlag, inRollWindow, riskShort,
    cycles: model.cycles.length, entryDebit: model.entryDebit,
    partial: shorts.length > 0 && shorts.reduce((s, l) => s + l.contracts, 0) < c,
    coveredContracts: shorts.reduce((s, l) => s + l.contracts, 0),
    state: closed ? 'Closed' : shorts.length ? (shorts.length > 1 ? `LEAPS + ${shorts.length} short calls open` : 'LEAPS + short call open') : 'LEAPS open · no short call'
  }
  x.glyphs = closed ? [] : GLYPHS.filter((g) => g.test(x)).map((g) => ({ ...g, toneKey: g.tone(x), text: g.tip(x) }))
  x.actions = pmccActions(x)
  x.awareness = x.glyphs.filter((g) => !PMCC_ACTION[g.glyph])
  const observedAt = preset.ivr ? new Date(ms(today) - preset.ivr.ageTradingDays * DAY + 20 * 3600000).toISOString() : null
  x.ivRank = preset.ivr ? { ...preset.ivr, observedAt } : null
  const input = { ivRank: x.ivRank, hasShortCall: shorts.length > 0, leapsInRollWindow: inRollWindow, earningsBeforeShortExpiry: earningsFlag, nextEarnings: preset.earnings ?? null }
  x.shortCtx = resolveContext(SHORT_CALL_CONTEXT, input)
  x.leapsCtx = resolveContext(LEAPS_CONTEXT, input)
  x.face = faceZone(x.ivRank, x.shortCtx)
  x.ivrInput = input
  return x
}

export function deriveWheel(w, today) {
  const m = w.kind ? markOf(w.bid, w.ask) : markOf(null, null)
  const pnl = w.kind && m.mid != null ? (w.fill - m.mid) * 100 * w.contracts : null
  const pctMax = pnl != null ? (w.fill - m.mid) / w.fill : null
  return {
    ...w, m, pnl, pctMax,
    exp: w.dte != null ? addDays(today, w.dte) : null,
    urgent: w.dte != null && w.dte <= EXPIRING_SOON_DTE,
    target: pctMax != null && pctMax * 100 >= PROFIT_TARGET,
    wide: m.mid != null && m.spread > WIDE_SPREAD * m.mid,
    maxProfit: w.kind ? w.fill * 100 * w.contracts : 0
  }
}

export const URGENCY_RANK = { high: 0, medium: 1, low: 2 }

export function shortLegRows(ticker, pmcc, leg) {
  // The engine's short-leg rules for one leg — shared by the wheel legs and the PMCC short (US-106).
  const time = leg.dte >= 0 && leg.dte <= EXPIRATION_IMMINENT_MAX_DTE
    ? [{ ticker, pmcc, urgency: 'high', code: 'EXPIRATION_IMMINENT', summary: `Expires in ${leg.dte} days at ${money(leg.strike)} strike` }]
    : leg.dte > EXPIRATION_IMMINENT_MAX_DTE && leg.dte <= MANAGEMENT_WINDOW_DTE
      ? [{ ticker, pmcc, urgency: 'medium', code: 'MANAGEMENT_WINDOW', summary: `${leg.dte} DTE remaining — review for roll or close` }]
      : []
  const profit = leg.captured != null && leg.captured * 100 >= PROFIT_TARGET
    ? [{ ticker, pmcc, urgency: 'low', code: 'PROFIT_TARGET', summary: `${pct(leg.captured)} of max profit captured — consider closing` }]
    : []
  return [...time, ...profit]
}

export function wheelRows(w) {
  // One wheel's queue rows; the queue and the wheel card's action strip both read these, so the strip mirrors the queue.
  if (!w.kind) return []
  const breach = w.phase === 'CC_OPEN' && w.price >= w.strike
  const near = w.phase === 'CSP_OPEN' && Math.abs((w.strike - w.price) / w.strike) * 100 <= 1
  return [
    ...shortLegRows(w.ticker, w.phase, { dte: w.dte, strike: w.strike, captured: w.pctMax }),
    ...(breach ? [{ ticker: w.ticker, pmcc: w.phase, urgency: 'medium', code: 'COVERED_CALL_BREACH', summary: `Stock is ${pct((w.price - w.strike) / w.strike)} above the ${money(w.strike)} call strike — shares may be called away` }] : []),
    ...(near ? [{ ticker: w.ticker, pmcc: w.phase, urgency: 'medium', code: 'STRIKE_PROXIMITY', itm: w.price < w.strike, summary: `Stock is ${pct(Math.abs((w.strike - w.price) / w.strike))} ${w.price >= w.strike ? 'above' : 'below'} the ${money(w.strike)} put strike` }] : [])
  ]
}

export const QUEUE_ACTION = {
  STRIKE_PROXIMITY: { glyph: '▲', word: 'STRIKE PROXIMITY', tone: (r) => (r.itm ? 'red' : 'gold'), rank: (r) => (r.itm ? 1 : 2) },
  COVERED_CALL_BREACH: { glyph: '▲', word: 'CALL BREACHED', tone: () => 'red', rank: () => 1 },
  EXPIRATION_IMMINENT: { glyph: '◷', word: 'EXPIRING', tone: () => 'gold', rank: () => 4 },
  MANAGEMENT_WINDOW: { glyph: '◷', word: 'MANAGEMENT WINDOW', tone: () => 'gold', rank: () => 5 }
}

export function queueAction(r) {
  const a = QUEUE_ACTION[r.code]
  return { glyph: a.glyph, word: a.word, tone: a.tone(r), rank: a.rank(r), reason: r.summary }
}

export function wheelActions(w) {
  // The wheel card's action tier is its own queue rows minus PROFIT_TARGET, which keeps the TARGET chip; ranks share the PMCC lane order.
  return rankActions(wheelRows(w).filter((r) => QUEUE_ACTION[r.code]).map(queueAction))
}

export function queueRows(x, wheels) {
  // The engine's registry, folded over the same legs the cards read. Short-leg rules ride on the PMCC short (US-106); ✕ and δ live on the card only.
  const rows = []
  if (!x.closed) {
    const s = x.nearShort
    if (s) {
      rows.push(...shortLegRows('XYZ', true, s))
      if (x.riskShort) rows.push({ ticker: 'XYZ', pmcc: true, urgency: 'medium', code: 'PMCC_ASSIGNMENT_RISK', summary: x.riskShort.itm
        ? `Stock is ${pct(-x.riskShort.pctBelow)} above the ${money(x.riskShort.strike)} short-call strike — in the money, LEAPS would have to cover`
        : `Stock is ${pct(x.riskShort.pctBelow)} below the ${money(x.riskShort.strike)} short-call strike — approaching assignment` })
      if (x.earningsFlag) rows.push({ ticker: 'XYZ', pmcc: true, urgency: 'medium', code: 'EARNINGS_PROXIMITY', summary: `Earnings ${x.daysToEarnings === 0 ? 'today' : x.daysToEarnings === 1 ? 'in 1 day' : `in ${x.daysToEarnings} days`} before your ${s.exp} expiration` })
    }
    if (x.inRollWindow) rows.push({ ticker: 'XYZ', pmcc: true, urgency: 'medium', code: 'PMCC_LEAPS_ROLL_WINDOW', summary: `LEAPS expires in ${x.leaps.dte} days at ${money(x.leaps.strike)} strike — review for roll or close` })
  }
  rows.push(...wheels.flatMap(wheelRows))
  return rows.sort((a, b) => URGENCY_RANK[a.urgency] - URGENCY_RANK[b.urgency] || a.ticker.localeCompare(b.ticker))
}

export function tier2Cells(x) {
  const l = x.leaps
  const s = x.nearShort
  const maxIfCalled = s ? (s.strike - l.strike - x.model.basis) * 100 * s.contracts : null
  return [
    ['Credits collected', money(x.creditsTotal), `${x.cycles} short-call cycle${x.cycles === 1 ? '' : 's'} · ${pct(x.recovered)} of LEAPS cost`],
    ['Breakeven', money(x.breakeven), `LEAPS ${money(l.strike)} + basis ${money(x.model.basis)} / share`],
    ['Max if called', s ? signed(maxIfCalled) : '—', s ? `called away at ${money(s.strike)}` : 'no short call open'],
    ['LEAPS delta', l.delta == null ? '—' : l.delta.toFixed(2), l.delta == null ? 'no greeks this poll' : l.delta < LEAPS_DELTA_FLOOR ? `below the ${LEAPS_DELTA_FLOOR.toFixed(2)} floor` : 'tracking as synthetic stock'],
    ['Short extrinsic', s && s.extrinsic != null ? money(s.extrinsic) : '—', s ? (s.itm ? 'ITM — assignment math is live' : 'OTM — all extrinsic') : 'no short call open'],
    ['LEAPS mark', l.m.mid == null ? '—' : money(l.m.mid), l.m.mid == null ? 'no snapshot this poll' : `${money(l.m.bid)} × ${money(l.m.ask)} · ${pct(l.m.spread / l.m.mid)} wide${l.wide ? ' · P&L unreliable' : ''}`],
    ['Entry net debit', money(x.entryDebit * 100 * x.contracts), `${money(x.entryDebit)} / share · opened ${x.model.start} · ${days(x.model.start, x.today)}d ago`]
  ]
}

        ) : (
          <div className={LEG_LINE}>
            <LegTag tag="·" />

            <span className="text-wb-text-muted">—</span>
            <span className="justify-self-end text-wb-text-muted">—</span>
          </div>
        )}
      </div>
      <div className="grid grid-cols-2 gap-px border-t border-wb-border bg-wb-border">
        <FootCell label="Cost basis">
          <div className="mt-[2px] font-wb-mono text-[13px] tabular-nums text-wb-sky">{money(w.basis)}</div>
          <div className={SUB}>/ share</div>
        </FootCell>
        <FootCell label="Premium collected">
          <div className="mt-[2px] font-wb-mono text-[13px] tabular-nums text-wb-green">{money(w.premium)}</div>
          <div className={SUB}>to date</div>
        </FootCell>
      </div>
    </CardShell>
  )
}

export function Derivation({ x }) {
  const l = x.leaps
  const expr = x.pnl != null
    ? `${x.legs.map((leg) => `${leg.side === 'SHORT' ? '− ' : ''}${money(leg.m.mid)} × 100 × ${leg.contracts}`).join(' ')} ${x.basisTotal < 0 ? `+ ${money(-x.basisTotal)}` : `− ${money(x.basisTotal)}`}`
    : x.closed ? 'closed — no live marks' : 'a mark is missing on an open leg'
  const iv = x.ivRank
  const rows = [
    ['Valuation date', 'America/New_York', x.today],
    ['Ledger', `LEAPS cost ${money(x.model.leapsCost)} − credits ${money(x.model.credits)} · ${x.cycles} cycle${x.cycles === 1 ? '' : 's'}`, `${money(x.model.basis)} / share`],
    ['LEAPS mid', l.m.mid == null ? 'no snapshot this poll' : `(${money(l.m.bid)} + ${money(l.m.ask)}) ÷ 2 · spread ${pct(l.m.spread / l.m.mid)}${l.wide ? ' · wide' : ''}`, l.m.mid == null ? '—' : money(l.m.mid)],
    ['Net P&L', expr, x.pnl == null ? '—' : signed(x.pnl)],
    ['% of basis', x.pnl == null ? '—' : x.pnlPct == null ? 'undefined — basis is not positive' : `${signed(x.pnl)} ÷ ${money(x.basisTotal)}`, x.pnlPct == null ? '—' : signedPct(x.pnlPct)],
    ...x.shorts.map((s, i) => [`Short ${x.shorts.length > 1 ? `${i + 1} ` : ''}% of max`, s.captured == null ? '—' : `(${money(s.fill)} − ${money(s.m.mid)}) ÷ ${money(s.fill)}`, s.captured == null ? '—' : `${s.captured < 0 ? '−' : ''}${pct(Math.abs(s.captured))}${s.target ? ' · TARGET' : ''}`]),
    ['Breakeven', `LEAPS ${money(l.strike)} + basis ${money(x.model.basis)}`, money(x.breakeven)],
    ['Premium recovered', `${money(x.model.credits)} ÷ ${money(x.model.leapsCost)} LEAPS cost`, pct(x.recovered)],
    ['Sort key', x.closed ? 'closed section' : x.nearShort ? 'nearest open short-call DTE' : 'LEAPS DTE — no short call', x.closed ? '—' : `${x.sortKey}d`],
    ['Action strip', x.actions.length ? x.actions.map((a) => `${a.glyph} ${a.word} (${a.owner}, ${a.tone}) — ${a.reason}`).join('  ·  ') : x.closed ? 'none — closed' : 'none — no action state open', x.actions.length ? `${x.actions[0].word}${x.actions.length > 1 ? ` +${x.actions.length - 1}` : ''}` : '—'],
    ['Awareness tags', x.awareness.length ? x.awareness.map((g) => `${g.glyph} ${g.name} (${g.owner}) — ${g.text}`).join('  ·  ') : x.closed ? 'none — closed' : 'none', x.awareness.map((g) => g.glyph).join(' ') || '—'],
    ['IV rank', iv === null ? 'no row for XYZ' : `${iv.value} · ${TIER_TITLE[iv.state]} · ${tradingDaysLabel(iv.ageTradingDays)} old · usable ${isUsableIvrState(iv.state) ? 'yes' : 'no'}`, x.closed ? '—' : x.face.word == null ? 'n/a · no word' : `${x.face.word} · ${x.face.tone}`],
    ['Short-call IV context', `SHORT_CALL_CONTEXT · ${x.shortCtx.rule} · short open ${x.ivrInput.hasShortCall ? 'yes' : 'no'} · ⚑ ${x.ivrInput.earningsBeforeShortExpiry ? 'yes' : 'no'}`, x.shortCtx.status],
    ['LEAPS IV context', `LEAPS_CONTEXT · ${x.leapsCtx.rule} · ◆ ${x.ivrInput.leapsInRollWindow ? 'yes' : 'no'}`, x.leapsCtx.status]
  ]
  return (
    <div className="overflow-hidden rounded-lg border border-wb-border bg-wb-bg-surface">
      <div className={`border-b border-wb-border px-[14px] py-[10px] font-wb-mono text-[10px] font-semibold uppercase tracking-[0.14em] ${x.preset.future || x.preset.projected ? 'text-wb-gold' : 'text-wb-text-muted'}`}>
        Derivation · {x.preset.label}{x.preset.future ? ' · not producible by the app yet' : ''}{x.preset.projected ? ' · cycles 10–12 are projected, not in the fixture' : ''}
      </div>
      <div className="grid grid-cols-[minmax(150px,200px)_1fr_auto] font-wb-mono text-[11.5px]">
        {rows.map(([k, e, v], i) => (
          <Fragment key={i}>
            <div className="border-t border-wb-border-subtle px-[14px] py-[7px] tracking-[0.04em] text-wb-text-muted">{k}</div>
            <div className="border-t border-wb-border-subtle px-[14px] py-[7px] text-wb-text-secondary">{e}</div>
            <div className="border-t border-wb-border-subtle px-[14px] py-[7px] text-right font-semibold tabular-nums text-wb-text-primary">{v}</div>
          </Fragment>
        ))}
      </div>
    </div>
  )
}

export const activeCount = (preset, wheels = WHEELS, pmcc = true) => wheels.length + (pmcc && !deriveCard(preset).closed ? 1 : 0)

export function PositionsGrid({ preset, wheels = WHEELS, pmcc = true, open = Boolean(preset.t2), onToggle = () => {} }) {
  // The Active + Closed card sections; the other Epic 09 mockups import this so the positions list is drawn once.
  // preset is one of PRESETS or a caller-built object of the same shape (today, leaps, cycles, q, ivr); wheels are raw WHEELS-shaped rows; pmcc=false leaves XYZ out.
  const x = pmcc ? deriveCard(preset) : null
  const derived = wheels.map((w) => deriveWheel(w, preset.today))
  const active = [
    ...derived.map((w) => ({ key: w.dte, el: <WheelCard key={w.id} w={w} /> })),
    ...(!x || x.closed ? [] : [{ key: x.sortKey, el: <PmccCard key={`xyz-${preset.key}`} x={x} open={open} onToggle={onToggle} /> }])
  ].sort((a, b) => (a.key == null) - (b.key == null) || a.key - b.key)
  const closedCards = [
    ...(x && x.closed ? [<PmccCard key={`xyz-closed-${preset.key}`} x={x} open={false} onToggle={() => {}} />] : []),
    <WheelCard key="tsla" w={deriveWheel(CLOSED_WHEEL, preset.today)} />
  ]
  return (
    <>
      <SectionHeader title="Active" />
      <div className="grid grid-cols-[repeat(auto-fill,minmax(400px,1fr))] gap-3 px-[24px] pb-[8px]">
        {active.map((a) => a.el)}
      </div>
      <SectionHeader title="Closed" />
      <div className="grid grid-cols-[repeat(auto-fill,minmax(400px,1fr))] gap-3 px-[24px] pb-[24px]">
        {closedCards}
      </div>
    </>
  )
}
