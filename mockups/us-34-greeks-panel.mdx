# US-34: Greeks Panel on Position Detail

_Adds a `Greeks` section to the position detail page rendered after `Open Leg` and before `Cost Basis`. Tier 1 (delta, theta) is prominent with severity color thresholds; Tier 2 (IV, vega) for context; Tier 3 (gamma) available but de-emphasised. Powered by the `OptionSnapshot.greeks` field returned by `useOptionSnapshots` from US-33._

## Acceptance Criteria States

- **csp-safe**: 16 DTE, $2.45 buffer (1.4%), delta 0.28 (green), theta $4.50/day → 21% yield
- **csp-moderate**: 16 DTE, delta 0.38 (gold) with "moderate assignment risk" tooltip
- **csp-danger**: 16 DTE, delta 0.52 (red) with "high assignment risk" tooltip
- **cc-moderate**: 16 DTE, CC delta 0.42 (gold) — CC threshold band is 0.35–0.50
- **csp-near-exp** _(new)_: 5 DTE, delta 0.38 fires **red** because the danger threshold tightens by 0.05 when DTE ≤ 7. Distance-to-strike row turns amber as buffer narrows.
- **csp-itm-urgent** _(new)_: 2 DTE, delta 0.55 — ITM banner pinned above Greeks ("URGENT: ITM with 2 days to expiration"). Distance to strike is negative.
- **unavailable**: provider returned no Greeks — muted placeholder, no error alert
- **holding-shares**: HOLDING_SHARES with no active leg — Greeks section omitted entirely

export const SEVERITY_COLOR = {
  normal: '#3fb950',
  warning: '#e6a817',
  danger: '#f85149',
  default: '#e6edf3',
  muted: '#6e7681'
}

export function deltaSeverity(absDelta, instrumentLabel, dte) {
  // Delta thresholds tighten by 0.05 when DTE ≤ 7 (gamma spikes overnight on 1–2% moves).
  const shift = dte <= 7 ? 0.05 : 0
  if (instrumentLabel === 'CC') {
    if (absDelta > 0.5 - shift) return 'danger'
    if (absDelta >= 0.35 - shift) return 'warning'
    return 'normal'
  }
  if (absDelta > 0.45 - shift) return 'danger'
  if (absDelta >= 0.3 - shift) return 'warning'
  return 'normal'
}

export function thetaYieldSeverity(yieldPct) {
  // Yield = (theta×DTE) / premium_collected × 100. Healthy when ≥50%.
  if (yieldPct >= 50) return 'normal'
  if (yieldPct >= 25) return 'default'
  return 'warning'
}

export function distanceSeverity(bufferPct) {
  // Positive buffer = OTM (good), negative = ITM (bad).
  if (bufferPct < 0) return 'danger'
  if (bufferPct < 1.5) return 'warning'
  return 'normal'
}

export function fmtMoney(n, signed) {
  const sign = signed && n > 0 ? '+' : n < 0 ? '−' : ''
  return `${sign}$${Math.abs(n).toFixed(2)}`
}
export function fmtPct(n, signed, digits = 1) {
  const sign = signed && n > 0 ? '+' : n < 0 ? '−' : ''
  return `${sign}${Math.abs(n).toFixed(digits)}%`
}

export function computeDistance(leg, underlyingPrice, instrumentLabel) {
  // CSP: stock above strike = buffer. CC: stock below strike = buffer.
  if (!leg || underlyingPrice == null) return null
  const raw = instrumentLabel === 'CC' ? leg.strike - underlyingPrice : underlyingPrice - leg.strike
  const pct = (raw / leg.strike) * 100
  return { dollars: raw, pct, severity: distanceSeverity(pct) }
}

export function GreeksPanel({ greeks, leg, instrumentLabel, underlyingPrice }) {
  if (greeks === null) {
    return (
      <SectionCard header="Greeks">
        <div
          style={{
            padding: '24px 20px',
            fontFamily: MONO,
            fontSize: '0.8rem',
            color: 'var(--wb-text-muted, #6e7681)',
            fontStyle: 'italic',
            display: 'flex',
            alignItems: 'center',
            gap: 8
          }}
        >
          <span style={{ opacity: 0.5 }}>◌</span>
          Greeks unavailable
        </div>
      </SectionCard>
    )
  }
  const colorFor = (sev) => SEVERITY_COLOR[sev] ?? SEVERITY_COLOR.default

// ---- DTE-aware derivations ----
const absDelta = Math.abs(greeks.delta)
const dte = leg.dte
const deltaSev = deltaSeverity(absDelta, instrumentLabel, dte)

// Theta: per-share → per-contract dollars/day. Yield = decay over remaining DTE / total premium.
const thetaPerContract = Math.abs(greeks.theta) _ 100
const premiumDollars = leg.premium _ leg.contracts _ 100
const thetaYieldPct = premiumDollars > 0 ? (thetaPerContract _ dte) / premiumDollars \* 100 : 0
const thetaSev = thetaYieldSeverity(thetaYieldPct)

// Gamma rises as DTE shrinks. Mark warning when |gamma| ≥ 0.04 within 7 DTE.
const gammaSev = dte <= 7 && Math.abs(greeks.gamma) >= 0.04 ? 'warning' : 'default'

// ---- Tier-1 banners ----
// ITM-near-expiration urgent: |Δ| > 0.50 with DTE ≤ 3.
const itmUrgent = dte <= 3 && absDelta > 0.50
// Soft notice when DTE ≤ 7 (thresholds tightened).
const dteTightened = dte <= 7 && !itmUrgent

const deltaTooltip = (() => {
if (deltaSev === 'danger') return 'High assignment risk'
if (deltaSev === 'warning')
return dte <= 7
? `Threshold tightened (DTE ≤ 7). Moderate ${instrumentLabel === 'CC' ? 'call-away' : 'assignment'} risk.`
: `Moderate ${instrumentLabel === 'CC' ? 'call-away' : 'assignment'} risk — consider rolling`
return `Low ${instrumentLabel === 'CC' ? 'call-away' : 'assignment'} risk`
})()

const thetaTooltip = `θ yield = (theta×DTE)/premium = ${thetaYieldPct.toFixed(0)}% — ${
    thetaSev === 'normal' ? 'healthy decay' : thetaSev === 'warning' ? 'anemic vs premium' : 'moderate decay'
  }`

return (

<SectionCard header="Greeks" headerRight={`${instrumentLabel} · ${dte}d · DTE-aware thresholds`}>

      {/* Urgent ITM-near-expiration banner */}
      {itmUrgent ? (
        <div
          role="alert"
          style={{
            display: 'flex',
            alignItems: 'center',
            gap: 10,
            padding: '10px 20px',
            background: '#f8514915',
            borderBottom: '1px solid #f8514940',
            color: '#f85149',
            fontFamily: MONO,
            fontSize: '0.75rem',
            letterSpacing: '0.04em',
            fontWeight: 600
          }}
        >
          <span style={{ fontSize: '0.95rem' }}>⚠</span>
          <span>URGENT: ITM with {dte} day{dte === 1 ? '' : 's'} to expiration</span>
          <span style={{ marginLeft: 'auto', fontWeight: 400, color: '#f85149bb', textTransform: 'none', letterSpacing: '0.02em' }}>
            Roll, close, or accept assignment
          </span>
        </div>
      ) : null}

      {/* DTE tightened notice */}
      {dteTightened ? (
        <div
          style={{
            display: 'flex',
            alignItems: 'center',
            gap: 8,
            padding: '6px 20px',
            background: '#e6a81710',
            borderBottom: '1px solid #e6a81730',
            color: '#e6a817',
            fontFamily: MONO,
            fontSize: '0.65rem',
            letterSpacing: '0.05em'
          }}
        >
          <span>◆</span>
          DTE ≤ 7 · delta thresholds shifted down 0.05 · gamma scrutiny on
        </div>
      ) : null}

      {/* Tier 1 — delta + theta */}
      <StatGrid minWidth={200}>
        <Stat
          label={`Delta${dte <= 7 ? ' · tight' : ''}`}
          value={absDelta.toFixed(2)}
          valueColor={colorFor(deltaSev)}
          sublabel={instrumentLabel === 'CC' ? 'call-away probability' : 'assignment probability'}
          size="1.4rem"
          weight={700}
          title={deltaTooltip}
        />
        <Stat
          label="Theta"
          value={fmtMoney(thetaPerContract) + '/day'}
          valueColor={colorFor(thetaSev)}
          sublabel={`${thetaYieldPct.toFixed(0)}% yield over ${dte}d`}
          size="1.4rem"
          weight={700}
          title={thetaTooltip}
        />
      </StatGrid>

      {/* Tier 2 — IV + vega + gamma */}
      <StatGrid minWidth={140}>
        <Stat label="IV" value={(greeks.iv * 100).toFixed(1) + '%'} sublabel="implied volatility" />
        <Stat label="Vega" value={fmtMoney(greeks.vega * 100)} sublabel="per 1% IV move" />
        <Stat
          label="Gamma"
          value={Math.abs(greeks.gamma).toFixed(3)}
          valueColor={colorFor(gammaSev)}
          sublabel="delta sensitivity"
          title={gammaSev === 'warning' ? 'Gamma elevated near expiration — delta can move 0.10–0.15 overnight on a 1–2% stock move' : undefined}
        />
      </StatGrid>
    </SectionCard>

)
}

export function OpenLegCard({ leg, underlyingPrice, instrumentLabel }) {
  if (!leg) return null
  const dteUrgent = leg.dte <= 7
  const dist = computeDistance(leg, underlyingPrice, instrumentLabel)
  const distColor = dist ? SEVERITY_COLOR[dist.severity] : undefined
  const distLabel = instrumentLabel === 'CC' ? 'Distance to Strike' : 'Distance to Strike'
  const distSublabel =
    dist && dist.dollars >= 0
      ? `${fmtPct(dist.pct, false)} buffer`
      : `${fmtPct(Math.abs(dist?.pct ?? 0), false)} ITM`
  return (
    <SectionCard header="Open Leg">
      <StatGrid minWidth={120}>
        <Stat label="Strike" value={fmtMoney(leg.strike)} valueColor="#e6a817" />
        <Stat label="Underlying" value={fmtMoney(underlyingPrice)} />
        <Stat
          label={distLabel}
          value={dist ? fmtMoney(dist.dollars, true) : '—'}
          valueColor={distColor}
          sublabel={distSublabel}
          title={
            dist && dist.severity === 'danger'
              ? 'ITM — stock has crossed the strike'
              : dist && dist.severity === 'warning'
                ? 'Buffer < 1.5% — monitor closely; small moves can flip ITM'
                : 'Healthy buffer'
          }
        />
        <Stat label="Expiration" value={leg.expiration} />
        <Stat label="DTE" value={`${leg.dte}d`} valueColor={dteUrgent ? '#e6a817' : undefined} />
        <Stat label="Contracts" value={leg.contracts} />
        <Stat label="Premium / Contract" value={fmtMoney(leg.premium)} valueColor="#3fb950" />
        <Stat label="Fill Date" value={leg.fillDate} />
      </StatGrid>
    </SectionCard>
  )
}
