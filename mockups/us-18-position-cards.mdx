# US-18: Position Cards Dashboard

_Display active positions as a responsive card grid with phase badge, ticker, strike, DTE countdown, premium collected, and effective cost basis._

## Acceptance Criteria States

- **multi-position**: Three active positions in different phases — CSP_OPEN, CC_OPEN, HOLDING_SHARES
- **dte-urgent**: Position with DTE ≤ 7 shown with gold urgency highlight
- **holding-no-leg**: HOLDING_SHARES position with no active option leg — strike and DTE display as "—"

export function MockPhaseBadge({ phase }) {

return (

<span style={{
      display: 'inline-flex', alignItems: 'center', gap: 5,
      fontSize: '0.7rem', fontWeight: 500, padding: '2px 9px', borderRadius: 4,
      background: cfg.color + '18', color: cfg.color, border: '1px solid ' + cfg.color + '30',

}

export function MockStat({ label, value, valueColor }) {
return (

export function PositionCard({ pos }) {
const isUrgent = pos.dte !== null && pos.dte <= 7
const dteColor = isUrgent ? 'var(--wb-gold, #e6a817)' : 'var(--wb-text-primary, #e2eaf2)'
const dteDisplay = pos.dte !== null ? `${pos.dte}d` : '—'
const strikeDisplay = pos.strike ? `${pos.instrumentLabel} $${pos.strike}` : '—'
const expDisplay = pos.expiration ?? '—'
const borderHover = 'rgba(230,168,23,0.3)'

return (

<div style={{
      background: 'var(--wb-bg-surface, #070a0e)',
      border: '1px solid var(--wb-border, #1e2a38)',
      borderRadius: 8, padding: 0, overflow: 'hidden',
      cursor: 'pointer', transition: 'border-color 0.15s',
      fontFamily: MONO,
    }}
onMouseOver={e => e.currentTarget.style.borderColor = borderHover}
onMouseOut={e => e.currentTarget.style.borderColor = 'var(--wb-border, #1e2a38)'} >
{/* ── Zone 1: Header — Ticker + Phase ── */}
{/* Real: <div><span>{ticker}</span><PhaseBadge phase={pos.phase} /></div> */}
<div style={{
        padding: '14px 16px 10px', display: 'flex', alignItems: 'center',
        justifyContent: 'space-between', borderBottom: '1px solid rgba(30,42,56,0.4)',
      }}>
<span style={{ fontSize: '1.1rem', fontWeight: 700, color: 'var(--wb-text-primary, #e2eaf2)', letterSpacing: '0.04em' }}>{pos.ticker}</span>
<MockPhaseBadge phase={pos.phase} />
</div>

      {/* ── Zone 2: Body — StatGrid ── */}
      {/* Real: <StatGrid minWidth={110} items={[{label:'Strike', value:…}, …]} /> */}
      <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 1, background: 'var(--wb-border, #1e2a38)' }}>
        <div style={{ padding: '10px 16px', background: 'var(--wb-bg-surface, #070a0e)' }}>
          <MockStat label="Strike" value={strikeDisplay} valueColor={pos.strike ? 'var(--wb-gold, #e6a817)' : '#4a5a6a'} />
        </div>
        <div style={{ padding: '10px 16px', background: 'var(--wb-bg-surface, #070a0e)' }}>
          <MockStat label="DTE" value={dteDisplay} valueColor={pos.dte !== null ? dteColor : '#4a5a6a'} />
        </div>
        <div style={{ padding: '10px 16px', background: 'var(--wb-bg-surface, #070a0e)' }}>
          <MockStat label="Premium" value={pos.premiumCollected !== '0.00' ? `+$${pos.premiumCollected}` : '$0.00'} valueColor="var(--wb-green, #3fb950)" />
        </div>
        <div style={{ padding: '10px 16px', background: 'var(--wb-bg-surface, #070a0e)' }}>
          <MockStat label="Cost Basis" value={`$${pos.effectiveCostBasis}`} valueColor="var(--wb-sky, #79c0ff)" />
        </div>
      </div>

      {/* ── Zone 3: Status Footer ── */}
      {/* Phase 1: expiration + urgency badge
         Phase 2 (US-22): distance-to-strike gauge replaces the urgency badge here.
         The slot is the same — only the content changes once live prices are available. */}
      <div style={{
        padding: '8px 16px', borderTop: '1px solid rgba(30,42,56,0.4)',
        fontSize: '0.7rem', color: pos.expiration ? 'var(--wb-text-secondary, #8899aa)' : '#4a5a6a',
        display: 'flex', alignItems: 'center', justifyContent: 'space-between',
      }}>
        <span>Exp: {expDisplay}</span>
        {isUrgent && (
          <span style={{
            fontSize: '0.6rem', fontWeight: 600, color: 'var(--wb-gold, #e6a817)',
            background: 'rgba(230,168,23,0.1)', padding: '1px 6px', borderRadius: 3,
            border: '1px solid rgba(230,168,23,0.25)',
          }}>EXPIRING SOON</span>
        )}
        <span style={{ fontSize: '0.65rem', color: '#4a5a6a' }}>View →</span>
      </div>
    </div>

)
}
