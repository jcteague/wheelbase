# Epic 09: PMCC Alert Surfaces — Final Layout

_The dashboard's management queue with **XYZ**, the fixture PMCC, in it beside two classic wheels — as it looks once the two PMCC alert rules have shipped. One rule watches the stock against the short-call strike (US-106), the other watches the calendar against the LEAPS (US-107). Every row, status and figure below is folded from the XYZ legs and the preset's valuation date and price; nothing is typed in. The dismissal panel (US-59) and the two threshold forms (US-57, US-58) are shown on the same rows and the same position so the reader can confirm they are unchanged._

## Which story owns what

The chrome is imported, not redrawn: `PageHeader`, `Breadcrumb`, `MarketStatusPill`, `NavItem`, `EnvironmentBadge`, `MarketDataStatusDot`, `SectionCard`, `UrgencyPill`, the `Sheet` parts and shadcn `Button` come from `src/renderer/src`. The classic-wheel rows are the shipped `ManagementQueueRow` itself.

| Region | Owner |
| -- | -- |
| Queue row for a `PMCC_ASSIGNMENT_RISK` alert: MED pill, PMCC badge in place of the wheel phase badge, summary, Review position | US-106 |
| Queue row for a `PMCC_LEAPS_ROLL_WINDOW` alert: same badge, a summary that begins "LEAPS" so the two XYZ rows read apart | US-107 |
| The PMCC badge itself (the teal `PMCC` pill, shared with the card and the detail hero) | US-106 introduces it in the queue; the strategy pill is US-101's / US-118's vocabulary |

| Classic-wheel rows, urgency ordering, the "N open alerts" count, Review position | US-51 — unchanged |
| Dismiss button and the confirmation panel on a PMCC row | US-59 — unchanged |

| Global alert defaults on Settings | US-57 — unchanged; US-106 / US-107 add no field |

## One queue, two clocks

A PMCC can hold two rows in the queue for the same ticker: one about the short call, one about the LEAPS. The queue does not grow a column for that. The badge says the position is a PMCC; the summary says which leg. The short-call rows use the wheel's own wording ("Expires in 2 days at $118.00 strike"), because the rule is the wheel's rule reading the PMCC's short leg; the LEAPS row is the only one that begins with "LEAPS". The 2% band and the 60-day window are fixed built-ins, referenced from the constants their stories define, and neither the per-position overrides nor the global defaults gain a field for them.

export const APPROACH_PERCENT = 2

export const WINDOW_DTE = 60

export const EXPIRATION_IMMINENT_MAX_DTE = 5

export const MANAGEMENT_WINDOW_DTE = 21

export const URGENCY_ORDER = { high: 0, medium: 1, low: 2 }

export const pctBelow = (price, strike) => Math.round(((strike - price) / strike) * 1000000) / 10000

export function inDays(n) {
  if (n === 0) return 'today'
  if (n === 1) return 'in 1 day'
  return `in ${n} days`
}

export function assignmentRiskSummary(price, strike) {
  const below = pctBelow(price, strike)
  const direction = price >= strike ? 'above' : 'below'
  const base = `Stock is ${Math.abs(below).toFixed(1)}% ${direction} the ${money(strike)} short-call strike`
  return direction === 'below' ? `${base} — approaching assignment` : `${base} — in the money, LEAPS would have to cover`
}

export function leapsRollWindowSummary(dte, strike) {
  return `LEAPS expires ${inDays(dte)} at ${money(strike)} strike — review for roll or close`
}

export function legsAt(today, leapsEvents, cycles) {
  const t = ms(today)
  const past = leapsEvents.filter((e) => ms(e.date) <= t)
  const last = past[past.length - 1]
  const leaps = last.kind === 'open' ? last : last.open
  const cycle = cycles.find((c) => ms(c.open.date) <= t && (!c.end || ms(c.end.date) > t)) ?? null
  const short = cycle
    ? [cycle.open, ...cycle.rolls.filter((r) => ms(r.date) <= t).map((r) => r.open)].pop()
    : null
  return { leaps, cycle, short }
}

export function evaluate(ctx) {
  const { leaps, cycle, short } = legsAt(ctx.today, ctx.leaps, ctx.cycles)
  const shortDte = short ? days(ctx.today, short.expiration) : null
  const leapsDte = ctx.unreadable ? null : days(ctx.today, leaps.expiration)
  const matches = []
  const skipped = []
  if (short) {
    if (shortDte >= 0 && shortDte <= EXPIRATION_IMMINENT_MAX_DTE) {
      matches.push({
        rule: 'EXPIRATION_IMMINENT',
        urgency: 'high',
        leg: 'short call',
        summary: `Expires in ${shortDte} days at ${money(short.strike)} strike`,
        since: dateMinus(short.expiration, EXPIRATION_IMMINENT_MAX_DTE)
      })
    } else if (shortDte > EXPIRATION_IMMINENT_MAX_DTE && shortDte <= MANAGEMENT_WINDOW_DTE) {
      matches.push({
        rule: 'MANAGEMENT_WINDOW',
        urgency: 'medium',
        leg: 'short call',
        summary: `${shortDte} DTE remaining — review for roll or close`,
        since: dateMinus(short.expiration, MANAGEMENT_WINDOW_DTE)
      })
    }
    if (ctx.price == null) {
      skipped.push({ rule: 'PMCC_ASSIGNMENT_RISK', reason: 'missing_underlying_price' })
    } else if (pctBelow(ctx.price, short.strike) <= APPROACH_PERCENT) {
      matches.push({
        rule: 'PMCC_ASSIGNMENT_RISK',
        urgency: 'medium',
        leg: 'short call',
        summary: assignmentRiskSummary(ctx.price, short.strike),
        since: ctx.today
      })
    }
  }
  if (ctx.unreadable) {
    skipped.push({ rule: 'PMCC_LEAPS_ROLL_WINDOW', reason: 'missing_leaps_dte' })
  } else if (leapsDte >= 0 && leapsDte <= WINDOW_DTE) {
    matches.push({
      rule: 'PMCC_LEAPS_ROLL_WINDOW',
      urgency: 'medium',
      leg: 'LEAPS',
      summary: leapsRollWindowSummary(leapsDte, leaps.strike),
      since: dateMinus(leaps.expiration, WINDOW_DTE)
    })
  }
  const distance = short && ctx.price != null ? pctBelow(ctx.price, short.strike) : null
  return { today: ctx.today, price: ctx.price, leaps, cycle, short, shortDte, leapsDte, matches, skipped, distance, unreadable: ctx.unreadable }
}

export function run(preset) {
  const ctx = {
    today: preset.today,
    price: preset.price ?? null,
    leaps: preset.leaps ?? BASE_LEAPS,
    cycles: preset.cycles ?? BASE_CYCLES,
    unreadable: Boolean(preset.unreadable)
  }
  const now = evaluate(ctx)
  const prev = preset.previous ? evaluate({ ...ctx, unreadable: false, ...preset.previous }) : null
  const keptOpen = prev ? prev.matches.filter((m) => now.skipped.some((s) => s.rule === m.rule)) : []
  const open = [...now.matches, ...keptOpen.map((m) => ({ ...m, kept: true }))].map((m) => {
    const was = prev ? prev.matches.find((p) => p.rule === m.rule) : null
    return { ...m, since: was ? was.since : m.since }
  })
  const resolved = prev ? prev.matches.filter((m) => !open.some((o) => o.rule === m.rule)) : []
  return { ...now, prev, open, resolved, keptOpen }
}

export function queueItems(model, preset) {
  const xyz = model.open
    .filter((m) => m.rule !== preset.dismissed)
    .map((m) => ({
      alertId: `xyz-${m.rule}`,
      positionId: 'xyz',
      ticker: 'XYZ',
      pmcc: true,
      leg: m.leg,
      rule: m.rule,
      kept: Boolean(m.kept),
      urgency: m.urgency,
      summary: m.summary,
      quickAction: 'Review position',
      triggeredAt: `${m.since}T09:31:00`
    }))
  return [...wheelRowsFor(model.today), ...xyz].sort(
    (a, b) =>
      URGENCY_ORDER[a.urgency] - URGENCY_ORDER[b.urgency] ||
      (a.triggeredAt < b.triggeredAt ? -1 : a.triggeredAt > b.triggeredAt ? 1 : 0)
  )
}

export function ruleStatus(model, rule, preset) {
  if (preset.dismissed === rule) return { label: 'DISMISSED · hidden', tone: 'text-wb-text-muted' }
  if (model.open.some((m) => m.rule === rule && m.kept)) return { label: 'OPEN · skipped this run', tone: 'text-wb-gold' }
  if (model.open.some((m) => m.rule === rule)) return { label: 'OPEN', tone: 'text-wb-gold' }
  if (model.resolved.some((m) => m.rule === rule)) return { label: 'RESOLVED', tone: 'text-wb-green' }
  if (rule === 'PMCC_ASSIGNMENT_RISK' && !model.short) return { label: 'NOT EVALUATED · no short call', tone: 'text-wb-text-muted' }
  return { label: 'NOT FIRING', tone: 'text-wb-green' }
}

export function cardPresetFor(preset) {
  // The XYZ card reads the same legs the rules read: the ledger cut at the valuation date, the preset's stock price, and no option marks (none are quoted here).
  const t = ms(preset.today)
  const leaps = (preset.leaps ?? BASE_LEAPS).filter((e) => ms(e.date) <= t)
  const cycles = (preset.cycles ?? BASE_CYCLES)
    .filter((c) => ms(c.open.date) <= t)
    .map((c) => ({ ...c, rolls: c.rolls.filter((r) => ms(r.date) <= t), end: c.end && ms(c.end.date) <= t ? c.end : null }))
  const q = { stock: preset.price ?? null, leapsBid: null, leapsAsk: null, leapsDelta: null, shortBid: null, shortAsk: null }
  return { key: preset.key, today: preset.today, leaps, cycles, q, ivr: null }
}

  const swatchShape = shape === 'dot' ? 'h-4 w-4 rounded-full border-2 border-wb-bg-base' : 'h-3 w-3 rounded-sm'
  return (
    <div className="contents">
      <span className="flex h-4 w-4 items-center justify-center">
        <span className={`inline-block shrink-0 ${swatchShape} ${swatchClass}`} />
      </span>
      <span className="text-wb-text-secondary">{label}</span>
      <span className={tone ?? 'text-wb-text-primary'}>{value}</span>
    </div>
  )
}

export function StrikeRuler({ model }) {
  const strike = model.short ? model.short.strike : 110
  const min = strike * 0.94
  const max = strike * 1.1
  const pct = (value) => `${Math.max(0, Math.min(((value - min) / (max - min)) * 100, 100))}%`
  const floor = strike * (1 - APPROACH_PERCENT / 100)
  const skipped = model.keptOpen.some((m) => m.rule === 'PMCC_ASSIGNMENT_RISK')
  const marker = model.price ?? (skipped && model.prev ? model.prev.price : null)
  const fires = model.open.some((m) => m.rule === 'PMCC_ASSIGNMENT_RISK' && !m.kept)
  const markerTone = skipped ? 'bg-wb-text-muted' : fires ? (model.distance <= 0 ? 'bg-wb-red' : 'bg-wb-gold') : 'bg-wb-green'
  const valueTone = fires ? (model.distance <= 0 ? 'text-wb-red' : 'text-wb-gold') : undefined
  return (
    <div className="flex flex-col gap-3 px-4 py-4">
      <div className="relative h-14">
        <div className="absolute inset-x-0 top-5 h-1.5 rounded-full bg-wb-bg-elevated" />
        {model.short ? (
          <div className="absolute top-5 h-1.5 bg-wb-gold-dim" style={{ left: pct(floor), width: `calc(${pct(strike)} - ${pct(floor)})` }} />
        ) : null}
        {model.short ? <div className="absolute top-5 h-1.5 rounded-r-full bg-wb-red-dim" style={{ left: pct(strike), right: 0 }} /> : null}
        {model.short ? <div className="absolute top-3 h-6 w-px bg-wb-gold" style={{ left: pct(floor) }} /> : null}
        {model.short ? <div className="absolute top-2 h-8 w-0.5 bg-wb-text-primary" style={{ left: pct(strike) }} /> : null}
        {marker === null ? null : (
          <div
            className={`absolute top-3.5 h-4 w-4 -translate-x-1/2 rounded-full border-2 border-wb-bg-base ${markerTone} ${skipped ? 'opacity-50' : ''}`}
            style={{ left: pct(marker) }}
          />
        )}
        <div className="absolute inset-x-0 top-10 flex justify-between text-[10px] text-wb-text-muted">
          <span>{money(min)}</span>
          <span>{money(max)}</span>
        </div>
      </div>
      <div className="mt-2 grid grid-cols-[auto_auto_1fr] items-center gap-x-3 gap-y-1.5 border-t border-wb-border pt-3 text-[10px] text-wb-text-muted">
        <LegendItem swatchClass="bg-wb-gold-dim" label="2% approach band" value={model.short ? `${money(floor)} to ${money(strike)}` : 'no short call'} />
        <LegendItem swatchClass="bg-wb-red-dim" label="In the money" value={model.short ? `${money(strike)} and above, no upper edge` : 'no short call'} />
        <LegendItem
          shape="dot"
          swatchClass={`${markerTone} ${skipped ? 'opacity-50' : ''}`}
          label={skipped ? 'Last known price' : 'Stock price'}
          value={marker === null ? 'unavailable' : money(marker)}
          tone={valueTone}
        />
      </div>
    </div>
  )
}

export function DteRuler({ model }) {
  const max = 200
  const pct = (d) => `${Math.max(0, Math.min((d / max) * 100, 100))}%`
  const skipped = model.keptOpen.some((m) => m.rule === 'PMCC_LEAPS_ROLL_WINDOW')
  const leapsMarker = skipped && model.prev ? model.prev.leapsDte : model.leapsDte
  const fires = model.open.some((m) => m.rule === 'PMCC_LEAPS_ROLL_WINDOW' && !m.kept)
  const leapsTone = skipped ? 'bg-wb-text-muted' : fires ? 'bg-wb-gold' : 'bg-wb-green'
  return (
    <div className="flex flex-col gap-3 px-4 py-4">
      <div className="relative h-14">
        <div className="absolute inset-x-0 top-5 h-1.5 rounded-full bg-wb-bg-elevated" />
        <div className="absolute top-5 h-1.5 rounded-l-full bg-wb-gold-dim" style={{ left: 0, width: pct(WINDOW_DTE) }} />
        <div className="absolute top-3 h-6 w-px bg-wb-gold" style={{ left: pct(WINDOW_DTE) }} />
        <div className="absolute top-2 h-8 w-0.5 bg-wb-text-primary" style={{ left: 0 }} />
        {model.short ? (
          <div className="absolute top-4 h-3 w-3 -translate-x-1/2 rotate-45 border border-wb-bg-base bg-wb-text-secondary" style={{ left: pct(model.shortDte) }} />
        ) : null}
        {leapsMarker === null ? null : (
          <div
            className={`absolute top-3.5 h-4 w-4 -translate-x-1/2 rounded-full border-2 border-wb-bg-base ${leapsTone} ${skipped ? 'opacity-50' : ''}`}
            style={{ left: pct(leapsMarker) }}
          />
        )}
        <div className="absolute inset-x-0 top-10 flex justify-between text-[10px] text-wb-text-muted">
          <span>today · {model.today}</span>
          <span>{max} days out</span>
        </div>
      </div>
      <div className="mt-2 grid grid-cols-[auto_auto_1fr] items-center gap-x-3 gap-y-1.5 border-t border-wb-border pt-3 text-[10px] text-wb-text-muted">
        <LegendItem swatchClass="bg-wb-gold-dim" label="Roll window" value={`0 to ${WINDOW_DTE} DTE, inclusive`} />
        <LegendItem
          shape="dot"
          swatchClass={`${leapsTone} ${skipped ? 'opacity-50' : ''}`}
          label={skipped ? 'LEAPS, last known' : 'LEAPS'}
          value={
            skipped
              ? `${money(model.leaps.strike)} · expiration unreadable · was ${model.prev.leapsDte} DTE`
              : `${money(model.leaps.strike)} · ${model.leaps.expiration} · ${model.leapsDte} DTE`
          }
          tone={fires ? 'text-wb-gold' : undefined}
        />
        <LegendItem
          swatchClass="rotate-45 bg-wb-text-secondary"
          label="Short call"
          value={model.short ? `${money(model.short.strike)} · ${model.short.expiration} · ${model.shortDte} DTE` : 'none open'}
        />
      </div>
    </div>
  )
}

  const ar = ruleStatus(model, 'PMCC_ASSIGNMENT_RISK', preset)
  const lrw = ruleStatus(model, 'PMCC_LEAPS_ROLL_WINDOW', preset)
  const rideAlong = model.open.filter((m) => m.rule === 'EXPIRATION_IMMINENT' || m.rule === 'MANAGEMENT_WINDOW')
  return (
    <SectionCard header={`Evaluation · XYZ · ${preset.today} · ${preset.scenario}`}>
      {preset.ruler ? (
        <StatGrid
          minWidth={150}
          items={[
            {
              label: 'Stock price',
              value: model.price === null ? <span className="text-wb-text-muted">unavailable</span> : money(model.price)
            },
            {
              label: 'Short call',
              value: model.short ? `${money(model.short.strike)} · ${model.short.expiration} · ${model.shortDte} DTE` : <span className="text-wb-text-muted">none open</span>
            },
            {
              label: 'LEAPS',
              value: model.unreadable ? (
                <span className="text-wb-text-muted">{money(model.leaps.strike)} · expiration unreadable</span>
              ) : (
                `${money(model.leaps.strike)} · ${model.leaps.expiration} · ${model.leapsDte} DTE`
              )
            },
            { label: 'PMCC_ASSIGNMENT_RISK', value: <span className={ar.tone}>{ar.label}</span> },
            { label: 'PMCC_LEAPS_ROLL_WINDOW', value: <span className={lrw.tone}>{lrw.label}</span> },
            {
              label: 'Short-leg rules riding along',
              value: rideAlong.length ? rideAlong.map((m) => m.rule).join(' · ') : <span className="text-wb-text-muted">none</span>
            }
          ]}
        />
      ) : null}
      {preset.ruler === 'price' ? <StrikeRuler model={model} /> : null}
      {preset.ruler === 'dte' ? <DteRuler model={model} /> : null}
      <div className="flex flex-col gap-3 px-4 pb-4 pt-4">
        <AlertBox variant={preset.tone}>{preset.note(model)}</AlertBox>
        {model.resolved.length ? (
          <div className="font-wb-mono text-[0.68rem] text-wb-text-muted">
            Resolved this run: {model.resolved.map((m) => `${m.rule} ("${m.summary}")`).join(' · ')}
          </div>
        ) : null}
        {model.keptOpen.length ? (
          <div className="font-wb-mono text-[0.68rem] text-wb-text-muted">
            Kept open on skip: {model.keptOpen.map((m) => m.rule).join(' · ')} — reason {model.skipped.map((s) => s.reason).join(', ')}
          </div>
        ) : null}
      </div>
    </SectionCard>
  )
}

export function Frame({ preset, onConfirmed }) {
  const model = run(preset)
  const items = queueItems(model, preset)
  const [confirming, setConfirming] = useState(
    preset.confirm ? { alertId: `xyz-${preset.confirm}`, ticker: 'XYZ' } : null
  )
  const page = preset.page ?? 'dashboard'
  return (
    <div className="flex flex-col gap-4">
      <Shell page={page}>
        {page === 'dashboard' ? (
          <DashboardPage
            cardPreset={cardPresetFor(preset)}
            wheels={wheelCardsFor(preset.today)}
            items={items}
            hasShort={Boolean(model.short)}
            confirming={confirming}
            onDismissClick={(alertId) => {
              const item = items.find((i) => i.alertId === alertId)
              setConfirming({ alertId, ticker: item ? item.ticker : 'XYZ' })
            }}
            onCancel={() => {
              setConfirming(null)
              if (preset.confirm) onConfirmed('in-the-money')
            }}
            onConfirm={() => {
              const target = confirming
              setConfirming(null)
              if (target && target.alertId === 'xyz-PMCC_ASSIGNMENT_RISK') onConfirmed('dismissed')
            }}
          />
        ) : null}
        {page === 'detail' ? <DetailPage /> : null}
        {page === 'settings' ? <SettingsPageMock /> : null}
      </Shell>
      <Evaluation model={model} preset={preset} />
    </div>
  )
}
