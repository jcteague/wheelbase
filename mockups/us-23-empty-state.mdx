# US-23: Dashboard Empty State

_Handle the zero-data state gracefully with onboarding copy and a CTA to create the first wheel. Also handles the "all wheels complete" variant and the filtered-empty-result inline message._

## Acceptance Criteria States

- **no-positions**: Zero positions exist — full empty state with illustration, heading "No positions yet", subheading, and "Open Your First Wheel" CTA
- **all-closed**: Active positions = 0 but closed positions exist — "All wheels complete" message with "Start a New Wheel" CTA
- **filtered-empty**: Positions exist but none match the selected phase filter — inline "No positions in this phase" message

export function FilterPillsStub({ activeFilter }) {
  const pills = [
    { label: 'All', count: 3, active: activeFilter === 'all' },
    { label: 'Sell Put', count: 0, active: false },
    { label: 'Holding', count: 1, active: false },
    { label: 'Sell Call', count: 2, active: activeFilter === 'cc' },
    { label: 'Closed', count: 0, active: false }
  ]
  return (
    <div style={{ display: 'flex', gap: 8, flexWrap: 'wrap' }}>
      {pills.map((p) => (
        <span
          key={p.label}
          style={{
            padding: '5px 12px',
            borderRadius: 6,
            fontFamily: MONO,
            fontSize: '0.7rem',
            border: '1px solid ' + (p.active ? 'rgba(230,168,23,0.4)' : '#1e2a38'),
            background: p.active ? 'rgba(230,168,23,0.1)' : 'transparent',
            color: p.active ? '#e6a817' : '#6a7a8a',
            display: 'inline-flex',
            alignItems: 'center',
            gap: 6
          }}
        >
          {p.label}
          <span
            style={{
              fontSize: '0.6rem',
              fontWeight: 700,
              background: p.active ? 'rgba(230,168,23,0.25)' : 'rgba(30,42,56,0.6)',
              color: p.active ? '#e6a817' : '#6a7a8a',
              padding: '0 5px',
              borderRadius: 8
            }}
          >
            {p.count}
          </span>
        </span>
      ))}
    </div>
  )
}
