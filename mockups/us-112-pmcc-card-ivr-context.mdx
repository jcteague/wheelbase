# US-112: IVR on the PMCC position card

_One number, two readings. The underlying's IV rank sits on the card face as a compact cell — numeral, freshness ring, one zone word — because "is this premium worth selling?" is the weekly question. Tier 2 reads the same number twice: what it means for the short call (the leg you sell, which likes rich IV) and what it means for the LEAPS (the leg you hold and roll, which pays up for it). Only a decision-usable reading gets a sentence; stale, expired, predates-earnings and missing readings look exactly as they do on the bench. The card is US-108's; this story adds the IVR cell and the two context cells and nothing else._

## Acceptance Criteria States

- **rich-fresh** — Background. `IVR 62` fresh, `rich`; tier 2: short `RICH`, LEAPS `PRICEY TO ROLL`
- **fair-fresh** — `45` → `fair`; both contexts `FAIR`
- **thin-fresh** — `22` → `thin` in gold; short `THIN`, LEAPS `CHEAP TO ROLL`
- **edge-30** — `30` is `fair`: the low zone is strictly below 30
- **edge-60-1** — `60.1` is `rich`: the high zone is strictly above 60 (60 itself is fair)
- **rich-earnings** — valuation 2026-11-16, earnings 2026-11-18 inside the short's window: ⚑ fires, the face word `rich` turns gold, and the short context reads `RICH · EVENT-DRIVEN`
- **thin-roll-window** — valuation 2027-07-20, LEAPS 59d (◆): low IV inside the window reads `ROLL NOW`
- **rich-roll-window** — same window, `62`: `PRICEY · WINDOW OPEN` — the window beats IV
- **leaps-only-rich** — no short call open: the short context speaks to the next sale
- **aging** — 2 trading days old: still usable, ring at three-quarters, contexts unchanged
- **stale** — 5 trading days old: muted `62 · 5d`, half ring, both contexts `—`
- **expired** — 12 trading days old: `exp`, quarter ring, both contexts `—`
- **predates-earnings** — observed before the last print: gold hollow ring, both contexts `—`
- **none** — no reading collected: `n/a` on the face and in both cells; every other figure unchanged
- **closed** — muted card in the Closed section: no IVR, no tier 2

Every IVR figure on the card is derived from the preset's reading (value, age, state) and the three
booleans US-108's card already computes (short call open, ◆ roll window, ⚑ earnings), so the mockup
and the acceptance criteria cannot drift apart. The derivation panel prints the resolution; the
registry panel prints both context tables in their order.

export const IVR_LOW_MAX = 30
export const IVR_HIGH_MIN = 60
export const ROLL_WINDOW_DTE = 60

export const d = (s) => new Date(s + 'T00:00:00Z')
export const dte = (from, to) => Math.round((d(to) - d(from)) / 86400000)
export const r2 = (n) => Math.round(n * 100) / 100
export function fmtMoney(n, signed) {
  if (n == null) return '—'
  const sign = signed ? (n > 0 ? '+' : n < 0 ? '-' : '') : n < 0 ? '-' : ''
  return sign + '$' + Math.abs(n).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })
}
export const fmtPct = (n, signed) => (signed && n > 0 ? '+' : n < 0 ? '-' : '') + Math.abs(n).toFixed(1) + '%'
export const fmtIvr = (v) => String(Number(v))
export const tradingDaysLabel = (n) => `${n} trading ${n === 1 ? 'day' : 'days'}`

export const TIER_TITLE = { fresh: 'Fresh', aging: 'Aging', stale: 'Stale', expired: 'Expired', predates_earnings: 'Predates earnings' }
export const isUsableIvrState = (s) => s === 'fresh' || s === 'aging'
export const TONE = { green: T.green, gold: T.gold, neutral: T.sec, muted: T.mut }

export function ivrZone(value) {
  const v = Number(value)
  return v < IVR_LOW_MAX ? 'low' : v > IVR_HIGH_MIN ? 'high' : 'mid'
}
export const ZONE_WORD = { low: 'thin', mid: 'fair', high: 'rich' }
export const ZONE_TONE = { low: 'gold', mid: 'neutral', high: 'green' }

export function unusable(ivRank) {
  if (ivRank === null) return { status: 'n/a', tone: 'muted', text: 'no IV rank collected' }
  if (ivRank.value === null) return { status: 'n/a', tone: 'muted', text: 'rank withheld — flat 52-week window' }
  const age = tradingDaysLabel(ivRank.ageTradingDays)
  if (ivRank.state === 'stale') return { status: 'Stale', tone: 'muted', text: `${age} old — collect a fresh reading before acting on it` }
  if (ivRank.state === 'expired') return { status: 'Expired', tone: 'muted', text: `last reading ${age} ago — collection has been failing` }
  if (ivRank.state === 'predates_earnings') return { status: 'Predates earnings', tone: 'muted', text: 'observed before the last print — IV has re-priced since' }
  return null
}

export function resolve(registry, input) {
  const blocked = unusable(input.ivRank)
  if (blocked) return { ...blocked, rule: 'unusable' }
  const i = { ...input, zone: ivrZone(input.ivRank.value) }
  const idx = registry.findIndex((r) => r.test(i))
  const r = registry[idx]
  return { status: r.status, tone: r.tone, text: r.text(i), rule: `row ${idx + 1}` }
}

export function faceZoneLabel(ivRank, shortCtx) {
  // The face word is the zone; its tone is the short-call read, so an event-driven rich reading is gold at a glance.
  if (ivRank === null || ivRank.value === null) return { word: 'n/a', tone: 'muted', zone: 'none' }
  if (!isUsableIvrState(ivRank.state)) return { word: TIER_TITLE[ivRank.state], tone: 'muted', zone: 'none' }
  const zone = ivrZone(ivRank.value)
  return { word: ZONE_WORD[zone], tone: shortCtx ? shortCtx.tone : ZONE_TONE[zone], zone }
}

export function derive(p) {
  const closed = p.status === 'CLOSED'
  const leaps = { ...p.leaps, dte: dte(p.valuation, p.leaps.exp) }
  const short = p.short ? { ...p.short, dte: dte(p.valuation, p.short.exp) } : null
  const basisTotal = r2(p.basisPerShare * 100)
  const premiumTotal = r2(p.creditsToDate * 100)
  const recovered = (p.creditsToDate / p.leapsCostToDate) * 100
  const pnl = r2((leaps.mid - (short ? short.mid : 0)) * 100 - basisTotal)
  const pnlPct = (pnl / basisTotal) * 100
  const shortPctMax = short ? ((short.fill - short.mid) / short.fill) * 100 : null
  const leapsInRollWindow = !closed && leaps.dte <= ROLL_WINDOW_DTE
  const daysToEarnings = p.earnings ? dte(p.valuation, p.earnings) : null
  const earningsBeforeShortExpiry = !!(short && daysToEarnings !== null && daysToEarnings >= 0 && daysToEarnings <= EARNINGS_PROXIMITY_MAX_DAYS && daysToEarnings <= short.dte)
  const pctBelow = short ? ((short.strike - p.price) / short.strike) * 100 : null
  const glyphs = []
  if (!closed) {
    if (leapsInRollWindow) glyphs.push({ g: '◆', n: 'Roll window', c: T.gold, t: `LEAPS inside the ${ROLL_WINDOW_DTE}-day roll window (US-107)` })
    if (short && pctBelow <= ASSIGNMENT_RISK_PCT_BELOW) glyphs.push({ g: '▲', n: 'Assignment risk', c: p.price >= short.strike ? T.red : T.gold, t: `Stock is ${Math.abs(pctBelow).toFixed(1)}% ${p.price >= short.strike ? 'above' : 'below'} the short-call strike (US-106)` })
    if (earningsBeforeShortExpiry) glyphs.push({ g: '⚑', n: 'Earnings', c: T.gold, t: `Earnings in ${daysToEarnings} days before your ${short.exp} expiration (US-56)` })
  }
  const input = { ivRank: p.ivr, hasShortCall: !!short, leapsInRollWindow, earningsBeforeShortExpiry, nextEarnings: p.earnings }
  const shortCtx = resolve(SHORT_CALL_CONTEXT, input)
  return { closed, leaps, short, basisTotal, premiumTotal, recovered, pnl, pnlPct, shortPctMax, glyphs, input,
    face: faceZoneLabel(p.ivr, shortCtx), shortCtx, leapsCtx: resolve(LEAPS_CONTEXT, input),
    sortKey: short ? short.dte : leaps.dte,
    state: closed ? 'Closed' : short ? 'LEAPS + short call open' : 'LEAPS open · no short call' }
}

export function FreshnessRing({ state, size = 14 }) {
  // Mirrors FreshnessRing.tsx: one fixed arc step per tier; predates_earnings is a hollow gold ring around a dot.
  const tier = { fresh: [1, T.green], aging: [0.75, T.sec], stale: [0.5, T.mut], expired: [0.25, T.mut], predates_earnings: [0, T.gold] }[state]
  const stroke = 2.25
  const r = (size - stroke) / 2
  const c = 2 * Math.PI * r
  const centre = size / 2
  return (
    <svg data-testid="freshness-ring" data-state={state} width={size} height={size} viewBox={`0 0 ${size} ${size}`} style={{ flexShrink: 0 }}>
      <circle cx={centre} cy={centre} r={r} fill="none" stroke={state === 'predates_earnings' ? T.gold : 'var(--wb-border, #21262d)'} strokeWidth={stroke} />
      {tier[0] > 0 ? <circle cx={centre} cy={centre} r={r} fill="none" stroke={tier[1]} strokeWidth={stroke} strokeDasharray={`${c * tier[0]} ${c}`} strokeLinecap="butt" transform={`rotate(-90 ${centre} ${centre})`} /> : null}
      {state === 'predates_earnings' ? <circle cx={centre} cy={centre} r={size / 6} fill={T.gold} /> : null}
    </svg>
  )
}

export function ivrTooltip(ivRank) {
  if (ivRank === null) return 'No IV rank collected'
  const age = tradingDaysLabel(ivRank.ageTradingDays)
  const v = fmtIvr(ivRank.value)
  switch (ivRank.state) {
    case 'predates_earnings': return `Predates earnings — observed at the close before the last earnings report. IV re-prices through a print, so this reading is unusable regardless of age.`
    case 'expired': return `Expired — last reading ${age} ago. Treated as no reading. Age this old means collection has been failing.`
    case 'stale': return `Stale — IV rank ${v}, ${age} old. Too old to trust: shown for context, never advised on.`
    case 'aging': return `Aging — IV rank ${v}, ${age} old. Still decision-usable; expect it to have drifted a few points.`
    default: return `Fresh — IV rank ${v}, ${age} old. Current as of the last session close.`
  }
}

export function IvrCell({ ivRank }) {
  // Mirrors IvrCell.tsx byte for byte in behaviour: n/a, exp, muted unusable, age suffix on aging/stale, ring, tooltip.
  if (ivRank === null) return <span className="ivrnum tip" data-ivr-state="empty" data-tip={ivrTooltip(null)} style={{ color: T.mut }}>n/a</span>
  const usable = isUsableIvrState(ivRank.state)
  const showsAge = ivRank.state === 'aging' || ivRank.state === 'stale'
  const expired = ivRank.state === 'expired'
  const value = expired ? 'exp' : fmtIvr(ivRank.value)
  return (
    <span className="ivrnum tip" data-testid="ivr-cell" data-ivr-state={ivRank.state} data-tip={ivrTooltip(ivRank)} style={{ color: usable ? T.pri : T.mut, fontWeight: expired ? 400 : 600 }}>
      <span>{value}{showsAge ? ` · ${ivRank.ageTradingDays}d` : ''}</span>
      <FreshnessRing state={ivRank.state} />
    </span>
  )
}
