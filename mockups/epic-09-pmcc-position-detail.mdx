# Epic 09: PMCC Position Detail — Final Layout

_The **XYZ** PMCC detail page as it looks once every Epic 09 story that touches it has shipped. One position, one history, one valuation date per preset; every figure on the page — lane statuses, P&L, guardrails, cycle nets, running basis, the six sheets' previews — is folded from the same legs and quotes, so the stories that share this page cannot disagree with each other._

## Which story owns what

The chrome is imported, not redrawn: `PageHeader`, `Breadcrumb`, `CollapsedDrawer`, `StatGrid`, `SheetPanel`, `IvrCell` (with its `FreshnessRing` and tooltip) and — on the `wheel` preset — `VerdictBlock` and `LegHistoryTable` come from `src/renderer/src`.

No detail-page story points outside this file any more. The per-story mockups `us-104-roll-pmcc-short-call.mdx` and `us-105-roll-pmcc-leaps.mdx` are design history; their states are carried here as presets on this file's XYZ ledger.

## The hierarchy, and why

A PMCC is two positions on two clocks. The page is ordered by how often a trader comes to it for each: **weekly** — does the short call need me? (left lane); **quarterly** — does the LEAPS need me? (right lane); **monthly** — is the position working? (guardrails); **rarely** — audit (the drawers). A single wheel-style verdict would have to average the two lanes, and the `in-the-money` preset is the case where they legitimately disagree.

## One ledger, three valuation dates for the roll and close sheets

US-104 and US-114 are written against the XYZ position on **2026-11-06** — and that day is already in `BASE_CYCLES`: cycle 2's roll (buy to close the $108 call at $3.00, sell the $112 call expiring 2026-12-18 at $2.40) is exactly the roll those stories describe. The Roll Short Call presets therefore use this ledger with cycle 2 still unrolled; the ledger folds to the stories' $21.20 basis and $101.20 effective share cost without any typed figure. US-105's roll is `BASE_LEAPS[1]` on **2027-07-09** (sell the $80 LEAPS at $26.00, buy the $85 · 2028-01-21 at $23.50), shown with the roll not yet recorded. US-120 starts from the `leaps-only` state on **2027-07-19**.

export function cycleShape(cycle, today) {
  const current = cycle.rolls.length ? cycle.rolls[cycle.rolls.length - 1].open : cycle.open
  const outcome = cycle.end ? cycle.end.outcome : 'open'
  const endDate = cycle.end ? cycle.end.date : today
  const rollNet = cycle.rolls.reduce((sum, r) => sum + (r.open.price - r.close.price), 0)
  const closeCost = cycle.end && cycle.end.outcome === 'closed' ? cycle.end.price : 0
  return {
    ...cycle,
    current,
    outcome,
    endDate,
    dteAtOpen: days(cycle.open.date, cycle.open.expiration),
    daysHeld: days(cycle.open.date, endDate),
    net: cycle.open.price + rollNet - closeCost,
    rollNet,
    closeCost
  }
}

export function buildModel(preset) {
  const today = preset.today
  const contracts = preset.contracts ?? 1
  const leapsEvents = preset.leaps ?? BASE_LEAPS
  const rawCycles = preset.cycles ?? (preset.patch ? preset.patch(BASE_CYCLES) : BASE_CYCLES)
  const cycles = rawCycles.map((c) => cycleShape(c, today))

  const ledger = []
  let leapsCost = 0
  let credits = 0
  const push = (date, label, group) => {
    ledger.push({ date, label, group, basis: leapsCost - credits })
  }
  const timed = []
  leapsEvents.forEach((e) => timed.push({ at: e.date, order: 0, kind: 'leaps', e }))
  cycles.forEach((c) => {
    timed.push({ at: c.open.date, order: 3, kind: 'cycle-open', c })
    c.rolls.forEach((r) => timed.push({ at: r.date, order: 2, kind: 'cycle-roll', c, r }))
    if (c.end) timed.push({ at: c.end.date, order: 1, kind: 'cycle-end', c })
  })
  timed
    .sort((a, b) => ms(a.at) - ms(b.at) || a.order - b.order)
    .forEach((t) => {
      if (t.kind === 'leaps') {
        if (t.e.kind === 'open') leapsCost += t.e.price
        else leapsCost += t.e.open.price - t.e.close.price
        push(t.at, t.e.kind === 'open' ? 'LEAPS bought' : 'LEAPS roll #1', 'leaps')
      } else if (t.kind === 'cycle-open') {
        credits += t.c.open.price
        push(t.at, `Cycle ${t.c.n} opened`, `cycle-${t.c.n}`)
      } else if (t.kind === 'cycle-roll') {
        credits += t.r.open.price - t.r.close.price
        push(t.at, `Cycle ${t.c.n} roll`, `cycle-${t.c.n}`)
      } else {
        if (t.c.end.outcome === 'closed') credits -= t.c.end.price
        push(t.at, `Cycle ${t.c.n} ${OUTCOME_LABEL[t.c.end.outcome].toLowerCase()}`, `cycle-${t.c.n}`)
      }
    })

  const basisAfter = (group) => {
    const rows = ledger.filter((l) => l.group === group)
    return rows.length ? rows[rows.length - 1].basis : null
  }

  const openLeaps = leapsEvents[leapsEvents.length - 1]
  const currentLeaps = openLeaps.kind === 'open' ? openLeaps : openLeaps.open
  const start = leapsEvents[0].date
  const closed = cycles.filter((c) => c.end)
  const openCycle = cycles.find((c) => !c.end)

  const gaps = []
  cycles.forEach((c, i) => {
    const prev = i === 0 ? null : cycles[i - 1]
    if (prev && prev.end) {
      const g = days(prev.end.date, c.open.date)
      if (g > 0) gaps.push({ after: prev.n, before: c.n, days: g, from: prev.end.date, to: c.open.date })
    }
  })
  const lastClosed = closed.length ? closed[closed.length - 1] : null
  if (!openCycle && lastClosed && days(lastClosed.end.date, today) > 0 && !preset.closeAll) {
    gaps.push({ after: lastClosed.n, before: null, days: days(lastClosed.end.date, today), from: lastClosed.end.date, to: today })
  }
  if (cycles.length === 0) {
    gaps.push({ after: null, before: null, days: days(start, today), from: start, to: today })
  } else if (days(start, cycles[0].open.date) > 0) {
    gaps.unshift({ after: null, before: cycles[0].n, days: days(start, cycles[0].open.date), from: start, to: cycles[0].open.date })
  }

  const totalDays = days(start, today)
  const uncovered = gaps.reduce((s, g) => s + g.days, 0)
  const realized = closed.reduce((s, c) => s + c.net, 0)
  const openCredit = openCycle ? openCycle.net : 0
  const leapsRollNet = leapsEvents
    .filter((e) => e.kind === 'roll')
    .reduce((s, e) => s + (e.close.price - e.open.price), 0)

  const flags = cycles
    .filter((c) => ms(c.current.expiration) > ms(currentLeaps.expiration))
    .map((c) => ({
      n: c.n,
      by: days(currentLeaps.expiration, c.current.expiration)
    }))

  return {
    today,
    contracts,
    start,
    leapsEvents,
    currentLeaps,
    cycles,
    gaps,
    ledger,
    basisAfter,
    openCycle,
    closedCount: closed.length,
    realized,
    openCredit,
    average: closed.length ? realized / closed.length : 0,
    leapsCost,
    credits,
    basis: leapsCost - credits,
    leapsRollNet,
    totalDays,
    uncovered,
    covered: totalDays - uncovered,
    flags,
    positionClosed: Boolean(preset.closeAll),
    closedAt: preset.closedAt,
    imported: Boolean(preset.imported),
    settlement: preset.settlement ?? null
  }
}

export function deriveCockpit(model, q) {
  const c = model.contracts
  const open = model.openCycle
  const eff = model.currentLeaps.strike + model.basis
  const leapsDte = days(model.today, model.currentLeaps.expiration)
  const shortDte = open ? days(model.today, open.current.expiration) : null
  const slack = open ? days(open.current.expiration, model.currentLeaps.expiration) : null
  const leapsMid = q ? mid(q.leapsBid, q.leapsAsk) : null
  const shortMid = q && open ? mid(q.shortBid, q.shortAsk) : null
  const captured = open && shortMid != null ? (open.net - shortMid) / open.net : null
  const distBelow = open && q ? (open.current.strike - q.stock) / open.current.strike : null
  const maxIfCalled = open ? (open.current.strike - eff) * 100 * c : null
  const wide = leapsMid != null && q.leapsAsk - q.leapsBid > 0.1 * leapsMid
  const intrinsic = q ? Math.max(0, q.stock - model.currentLeaps.strike) : null
  const extrinsic = leapsMid != null ? leapsMid - intrinsic : null

  let pnl = null
  if (model.settlement && model.settlement.path === 'sold') {
    pnl = model.settlement.leapsSale * 100 * c - (model.settlement.fees ?? 0) - model.basis * 100 * c
  } else if (model.settlement) {
    pnl =
      (model.settlement.leapsSale - model.settlement.coverPrice + model.cycles[model.cycles.length - 1].current.strike) *
        100 *
        c -
      model.basis * 100 * c
  } else if (leapsMid != null && (!open || shortMid != null)) {
    pnl = (leapsMid - (open ? shortMid : 0)) * 100 * c - model.basis * 100 * c
  }
  const pnlPct = pnl != null && model.basis > 0 ? pnl / (model.basis * 100 * c) : null

  let shortStatus
  if (model.positionClosed)
    shortStatus = {
      label: 'CLOSED',
      tone: 'muted',
      reason: model.settlement && model.settlement.path === 'sold' ? `no short call was open · position sold to close ${model.closedAt}` : `assigned ${model.closedAt} · settled`
    }
  else if (!open) {
    const last = model.cycles[model.cycles.length - 1]
    const idle = last && last.end ? days(last.end.date, model.today) : days(model.start, model.today)
    shortStatus = { label: 'NO SHORT CALL', tone: 'gold', reason: `${idle} days uncovered · sell one to resume income` }
  } else if (open.current.strike < eff)
    shortStatus = {
      label: 'BELOW BREAKEVEN',
      tone: 'red',
      reason: `strike ${money(open.current.strike)} is ${money(eff - open.current.strike)} under your effective share cost · call-away locks in ${signed(maxIfCalled)}${q && q.stock >= open.current.strike ? ' · stock is through the strike' : ''}`
    }
  else if (q && q.stock >= open.current.strike)
    shortStatus = {
      label: 'ASSIGNMENT RISK',
      tone: 'red',
      reason: `in the money by ${money(q.stock - open.current.strike)}${captured != null ? ` · ${pct(captured)} of max` : ''} · LEAPS would have to cover`
    }
  else if (distBelow != null && distBelow <= 0.02)
    shortStatus = {
      label: 'ASSIGNMENT RISK',
      tone: 'gold',
      reason: `${pct(distBelow)} below strike${captured != null ? ` · ${pct(captured)} of max captured` : ''}`
    }
  else if (shortDte < 0)
    shortStatus = { label: 'EXPIRED · RECORD IT', tone: 'gold', reason: `expired ${open.current.expiration} below the strike · record the expiration` }
  else if (captured != null && captured >= 0.5)
    shortStatus = { label: 'TAKE PROFIT', tone: 'gold', reason: `${pct(captured)} of max captured · ${pct(distBelow)} below strike` }
  else if (shortDte <= 7)
    shortStatus = { label: 'EXPIRING', tone: 'gold', reason: `${shortDte}d left${captured != null ? ` · ${pct(captured)} of max captured` : ''} · ${pct(distBelow)} below strike` }
  else
    shortStatus = {
      label: 'HOLD',
      tone: 'green',
      reason: `${captured != null ? `${pct(captured)} of max captured · ` : ''}${distBelow != null ? `${pct(distBelow)} below strike · ` : ''}let theta work`
    }

  let leapsStatus
  if (model.positionClosed) leapsStatus = { label: 'CLOSED', tone: 'muted', reason: `sold to close ${model.settlement.date} at ${money(model.settlement.leapsSale)}` }
  else if (q && q.stock <= model.currentLeaps.strike)
    leapsStatus = {
      label: 'BROKEN',
      tone: 'red',
      reason: `stock ${money(model.currentLeaps.strike - q.stock)} below the ${money(model.currentLeaps.strike)} strike · the LEAPS is all extrinsic`
    }
  else if (q && q.leapsDelta != null && q.leapsDelta < 0.6)
    leapsStatus = { label: 'DELTA DECAYING', tone: 'gold', reason: `Δ ${q.leapsDelta.toFixed(2)} below 0.60 · no longer tracking as synthetic stock` }
  else if (leapsDte <= 60)
    leapsStatus = { label: 'ROLL WINDOW', tone: 'gold', reason: `${leapsDte}d · roll while it still has time value to sell` }
  else
    leapsStatus = {
      label: 'HEALTHY',
      tone: 'green',
      reason: `tracking as synthetic stock${q ? ` · ${pct((q.stock - model.currentLeaps.strike) / model.currentLeaps.strike)} above strike` : ''}`
    }

  let actions = []
  if (!model.positionClosed) {
    if (!open) actions = ['Sell Short Call →', 'Roll LEAPS →', 'Close Position →']
    else if (shortDte < 0) actions = ['Record Expiration →', 'Record Assignment →', 'Roll LEAPS →']
    else actions = ['Roll Short Call →', 'Close Short Early →', 'Record Assignment →', 'Roll LEAPS →']
  }

  return { eff, leapsDte, shortDte, slack, leapsMid, shortMid, captured, distBelow, maxIfCalled, wide, intrinsic, extrinsic, pnl, pnlPct, shortStatus, leapsStatus, actions }
}

export function AssignmentSheet({ model, k, q, onClose }) {

  const c = model.contracts
  const shares = 100 * c
  const strike = open.current.strike
  const L = model.currentLeaps.strike
  const sale = parseFloat(leapsSale)
  const buy = parseFloat(sharePrice)
  const basisTotal = model.basis * 100 * c
  const deliver = strike * shares
  const exerciseCash = deliver - L * shares
  const sellCoverCash = (sale > 0 ? sale : 0) * shares - (buy > 0 ? buy : 0) * shares + deliver
  const coverCost = deliver - (buy > 0 ? buy : 0) * shares
  const coverPerShare = buy > 0 ? buy - strike : 0
  const cycleNetKeep = open.net - coverPerShare
  const basisKeep = model.basis + coverPerShare
  const realized = settle === 'exercise' ? exerciseCash - basisTotal : settle === 'sell-cover' ? sellCoverCash - basisTotal : null
  const belowEff = strike < k.eff
  const errors = {}
  if (settle === 'sell-cover' && !(sale > 0)) errors.sale = 'LEAPS sale price must be greater than zero'
  if (settle !== 'exercise' && !(buy > 0)) errors.buy = 'Share purchase price must be greater than zero'

      <SheetBody>
        <SectionCard header="Assigned short call">
          <StatGrid
            minWidth={100}
            items={[
              { label: 'Strike', value: <span className="text-wb-gold">{money(strike)}</span> },
              { label: 'Sold', value: `${open.open.date} at ${money(open.open.price)}` },
              { label: 'Delivered', value: `${shares} sh at ${money(strike)}` },
              { label: 'Received', value: <span className="text-wb-green">+{money(deliver)}</span> }
            ]}
          />
        </SectionCard>
        <SectionCard header="Your LEAPS">
          <StatGrid
            minWidth={100}
            items={[
              { label: 'Strike', value: <span className="text-wb-gold">{money(L)}</span> },
              { label: 'DTE', value: k.leapsDte },
              { label: 'Mid', value: `${money(k.leapsMid)} (${money(q.leapsBid)} × ${money(q.leapsAsk)})` },
              { label: 'Intrinsic · extrinsic', value: `${money(k.intrinsic)} · ${money(k.extrinsic)}` }
            ]}
          />
        </SectionCard>

        <div className="flex flex-col gap-2">
          <span className="font-wb-mono text-xs font-semibold uppercase tracking-widest text-wb-text-secondary">Settlement</span>
          <Option id="exercise" title="Exercise the LEAPS" note={`forfeits ${money(k.extrinsic * shares)} extrinsic · position closes`} />
          <Option id="sell-cover" title="Sell the LEAPS and buy shares to deliver" note={`captures the extrinsic · better than exercising by ${money(k.extrinsic * shares)} · position closes`} />
          <Option id="cover-keep" title="Buy shares to deliver, keep the LEAPS" note="cycle nets a loss · position continues LEAPS-only" />
        </div>

        {settle !== 'exercise' && (
          <div className="grid grid-cols-2 gap-3">
            {settle === 'sell-cover' && (
              <Field label="LEAPS sale price / share" htmlFor="as-sale" error={errors.sale}>
                <Money id="as-sale" value={leapsSale} onChange={setLeapsSale} error={errors.sale} />

            </Field>

            </Field>
          </div>

                <SummaryRow label={`Deliver at ${money(strike)}`} value={`+${money(deliver)}`} />
                <SummaryRow label="Settlement cash" value={signed(exerciseCash)} />
                <SummaryRow label="Less cost basis" value={`−${money(basisTotal)}`} />
              </>
            )}
            {settle === 'sell-cover' && (
              <>
                <SummaryRow label="Sell LEAPS" value={`+${money((sale > 0 ? sale : 0) * shares)}`} />
                <SummaryRow label={`Buy ${shares} sh`} value={`−${money((buy > 0 ? buy : 0) * shares)}`} />

                <SummaryRow label="Settlement cash" value={signed(sellCoverCash)} />
                <SummaryRow label="Less cost basis" value={`−${money(basisTotal)}`} />
              </>
            )}
            {settle === 'cover-keep' && (
              <>
                <SummaryRow label={`Buy ${shares} sh`} value={`−${money((buy > 0 ? buy : 0) * shares)}`} />
                <SummaryRow label={`Deliver at ${money(strike)}`} value={`+${money(deliver)}`} />
                <SummaryRow label="Cost to cover" value={signed(coverCost)} />
                <SummaryRow label={`Cycle ${open.n} net`} value={`${signed(cycleNetKeep)} / share (${signed(cycleNetKeep * shares)})`} />

            {realized != null && (
              <div className="mt-1 flex items-baseline justify-between border-t border-wb-border pt-2">
                <span className="font-wb-mono text-[0.75rem] text-wb-text-muted">Realized P&L</span>
                <span className="font-wb-mono text-lg font-bold" style={{ color: realized >= 0 ? TONE.green : TONE.red }}>
                  {signed(realized)}
                </span>
              </div>
            )}
          </div>
        </SectionCard>
        {belowEff && (
          <AlertBox variant="error">
            The short strike sat below your effective share cost of {money(k.eff)} — this assignment locks in a loss on the diagonal.
          </AlertBox>
        )}
        {settle === 'exercise' && k.extrinsic > 0 && (
          <AlertBox variant="warning">Forfeits {money(k.extrinsic * shares)} of extrinsic value. Selling the LEAPS and buying shares captures it.</AlertBox>
        )}
      </SheetBody>
      <SheetFooter>
        <FormButton label="Cancel" variant="secondary" onClick={onClose} />
        <FormButton label="Confirm Assignment" disabled={Object.keys(errors).length > 0} />
      </SheetFooter>
    </SheetPanel>
  )
}

export function rollLabel(from, to, fmt) {
  const direction = to.strike > from.strike ? 'Up' : to.strike < from.strike ? 'Down' : null
  const timing = ms(to.expiration) > ms(from.expiration) ? 'Out' : ms(to.expiration) < ms(from.expiration) ? 'In' : null
  const parts = [direction, timing].filter(Boolean)
  if (parts.length === 0) return null
  const strikeText = direction ? `$${from.strike} to $${to.strike} strike` : `same $${from.strike} strike`
  const dateText = timing ? `${fmt(from.expiration)} to ${fmt(to.expiration)} expiration` : `same ${fmt(from.expiration)} expiration`
  return `Roll ${parts.join(' & ')}: ${strikeText}, ${dateText}`
}

export const ivrZone = (value) => (value > 60 ? 'rich' : value >= 30 ? 'fair' : 'thin')

export const ivrUsable = (r) => r != null && (r.state === 'fresh' || r.state === 'aging')

export const earningsInsideWindow = (nextEarnings, expiration) =>
  Boolean(nextEarnings && expiration && ms(nextEarnings) <= ms(expiration))

export const UNUSABLE_CONTEXT = {
  stale: (r) => ({ title: 'Stale', status: 'Stale · —', sentence: `${r.ageTradingDays} trading days old — collect a fresh reading before acting on it` }),
  expired: (r) => ({ title: 'Expired', status: 'Expired · —', sentence: `last reading ${r.ageTradingDays} trading days ago — collection has been failing` }),
  predates_earnings: () => ({ title: 'Predates earnings', status: 'Predates earnings · —', sentence: 'observed before the last print — IV has re-priced since' })
}

export const SHORT_CALL_CONTEXT = [
  { test: (i) => i.zone === 'rich' && i.eventDriven, status: 'RICH · EVENT-DRIVEN', tone: 'gold', sentence: (i) => `IV re-prices after the ${i.nextEarnings} print — don't anchor on it` },
  { test: (i) => i.zone === 'rich', status: 'RICH', tone: 'green', sentence: () => 'premium is rich — a roll up-and-out collects more than usual' },
  { test: (i) => i.zone === 'thin', status: 'THIN', tone: 'gold', sentence: () => 'premium is thin — the next cycle pays less; consider a longer DTE' },
  { test: () => true, status: 'FAIR', tone: 'neutral', sentence: () => 'premium is fair — run the usual cycle' }
]

export function shortCallContext({ ivRank, nextEarnings, expiration }) {
  // US-112's SHORT_CALL_CONTEXT registry with hasShortCall: true — the rows that need no short call cannot fire in a roll sheet.
  if (!ivRank) return { status: 'n/a', tone: 'muted', sentence: 'no IV rank collected', zoneWord: null, zoneTone: 'muted' }
  if (!ivrUsable(ivRank)) return { ...UNUSABLE_CONTEXT[ivRank.state](ivRank), tone: 'muted', zoneWord: null, zoneTone: 'muted' }
  const input = { zone: ivrZone(Number(ivRank.value)), eventDriven: earningsInsideWindow(nextEarnings, expiration), nextEarnings }
  const row = SHORT_CALL_CONTEXT.find((r) => r.test(input))
  return { status: row.status, tone: row.tone, sentence: row.sentence(input), zoneWord: input.zone, zoneTone: row.tone }
}

export function Ruler({ today, marks, fillTo, blocked, note }) {
  // marks: { date | null, label, value, bar (tick classes) } — the `tall` mark is the proposed contract and takes the ruler's tone.
  const t0 = ms(today)
  const dated = marks.filter((m) => m.date)
  const t1 = Math.max(...dated.map((m) => ms(m.date)))
  const at = (d) => `${Math.max(0, Math.min(100, ((ms(d) - t0) / (t1 - t0)) * 100)).toFixed(2)}%`

            <SummaryRow label="Effective share cost" value={money(k.eff)} />
          </div>
        </SectionCard>

        <div className="grid grid-cols-2 gap-3">
          <Field label="Sale price / share" htmlFor="cp-sale" error={errors.sale} hint={k.leapsMid != null ? 'defaults to the mid' : 'no quote — type your fill'}>
            <Money id="cp-sale" value={sale} onChange={setSale} error={errors.sale} />
          </Field>
          <Field label="Fill date" htmlFor="cp-fill" error={errors.fill}>
            <NumberInput id="cp-fill" value={fill} hasError={Boolean(errors.fill)} onChange={(e) => setFill(e.target.value)} />
          </Field>
          <Field label="Fees" htmlFor="cp-fees" error={errors.fees}>
            <Money id="cp-fees" value={fees} onChange={setFees} error={errors.fees} />
          </Field>
        </div>

        <SectionCard header="Result" headerVariant="emphasized">
          <div className="flex flex-col gap-1.5 p-4">
            {final == null ? (
              <div className="py-2 text-center text-xs text-wb-text-muted">Enter the sale price to preview the result.</div>
            ) : (
              <>
                <SummaryRow label={`Proceeds · ${money(saleN)} × ${shares} sh`} value={`+${money(proceeds)}`} />
                <SummaryRow label="Fees" value={`−${money(feeTotal)}`} />
                <SummaryRow label="Cost basis" value={`−${money(basisTotal)}`} />
                <div className="mt-1 flex items-baseline justify-between border-t border-wb-border pt-2">
                  <span className="text-[0.75rem] text-wb-text-muted">{'Final P&L'}</span>
                  <span className={`text-lg font-bold ${losing ? 'text-wb-red' : 'text-wb-green'}`}>
                    {signed(final)}{' '}
                    <span className="text-xs font-normal text-wb-text-muted">{pctBasis == null ? '' : `${pctBasis >= 0 ? '+' : ''}${pct(pctBasis)} of basis`}</span>
                  </span>
                </div>
                <div className="text-[11px] text-wb-text-muted">
                  LEAPS cost {money(model.leapsCost)} − credits {money(model.credits)} = cost basis {money(model.basis)} / share · matches the Cost basis drawer
                </div>
                {losing && (
                  <div className="text-[11px] leading-relaxed text-wb-gold">
                    The LEAPS is being sold {money(model.leapsCost - saleN)} / share below its cost to date; short-call credits of {money(model.credits)} recover all but {money(-final / shares)}.
                  </div>
                )}
              </>
            )}
          </div>
        </SectionCard>
        <AlertBox variant="warning">This closes the position. Full leg history is preserved.</AlertBox>
      </SheetBody>
      <SheetFooter>
        <FormButton label="Cancel" variant="secondary" onClick={onClose} />
        <FormButton label="Confirm Close" disabled={!valid} />
      </SheetFooter>
    </SheetPanel>
  )
}

export function RecordedBanner({ model, kind }) {
  const c = model.contracts
  if (kind === 'short') {
    const cycle = model.openCycle
    const r = cycle.rolls[cycle.rolls.length - 1]
    const net = r.open.price - r.close.price
    return (
      <AlertBox variant="success">
        Rolled — bought to close XYZ {money(r.close.strike)} call {r.close.expiration} at {money(r.close.price)}, sold XYZ {money(r.open.strike)} call {r.open.expiration} at{' '}
        {money(r.open.price)}. {net < 0 ? 'Net debit' : 'Net credit'} {money(Math.abs(net))} per share ({money(Math.abs(net) * 100 * c)}). Both legs share one roll chain · cost basis{' '}
        {money(model.basis + net)} → {money(model.basis)} · US-103 history shows one line "Short-call roll #{cycle.rolls.length} {net < 0 ? 'debit' : 'credit'}" of {money(Math.abs(net))}.
      </AlertBox>
    )
  }
  const e = model.leapsEvents[model.leapsEvents.length - 1]
  const net = e.close.price - e.open.price
  return (
    <AlertBox variant="success">
      Rolled — sold to close XYZ {money(e.close.strike)} call {e.close.expiration} at {money(e.close.price)}, bought to open XYZ {money(e.open.strike)} call {e.open.expiration} at{' '}
      {money(e.open.price)}. {net < 0 ? 'Net debit' : 'Net credit'} {money(Math.abs(net))} per share ({money(Math.abs(net) * 100 * c)}). LEAPS cost to date {money(model.leapsCost + net)} →{' '}
      {money(model.leapsCost)} · cost basis {money(model.basis + net)} → {money(model.basis)} · effective share cost re-anchored to {money(e.open.strike)} + {money(model.basis)} ={' '}
      {money(e.open.strike + model.basis)}.
    </AlertBox>
  )
}
